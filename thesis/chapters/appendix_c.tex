\section{Symptom Tokenization}
%==============================================================================

\begin{lstlisting}[language=Python, caption=Token Sequence Construction]
def build_tokens(X, seq_len=25, lab_count=15):
    n_samples, n_features = X.shape
    tokens = np.zeros((n_samples, seq_len, 4), dtype=np.float32)
    for i in range(n_samples):
        valid = np.where(~np.isnan(X[i]) & (X[i] >= 0) & (X[i] <= 1))[0]
        for j, idx in enumerate(valid[:seq_len]):
            tokens[i, j, 0] = idx + 1  # symptom_id
            tokens[i, j, 1] = float(X[i, idx])  # intensity
            tokens[i, j, 2] = 0.0  # time_delta
            tokens[i, j, 3] = 0.0 if idx < lab_count else 1.0  # modality
    return tokens
\end{lstlisting}

\section{Prototypical Head}

\begin{lstlisting}[language=Python, caption=Prototype Computation and Classification]
def compute_prototypes(support_embeddings, support_labels):
    classes = torch.unique(support_labels)
    prototypes = []
    for c in classes:
        mask = support_labels == c
        proto = support_embeddings[mask].mean(dim=0)
        prototypes.append(proto)
    return torch.stack(prototypes)

def classify_query(query_embeddings, prototypes, temperature=10.0):
    logits = F.cosine_similarity(
        query_embeddings.unsqueeze(1),
        prototypes.unsqueeze(0),
        dim=-1
    ) * temperature
    return F.softmax(logits, dim=-1)
\end{lstlisting}

\section{Focal Loss}
%==============================================================================

\begin{lstlisting}[language=Python, caption=Focal Loss Implementation]
def focal_loss(logits, targets, gamma=2.0, alpha=None):
    ce = F.cross_entropy(logits, targets, reduction='none')
    pt = torch.exp(-ce)
    focal = ((1 - pt) ** gamma) * ce
    if alpha is not None:
        alpha_t = alpha[targets]
        focal = alpha_t * focal
    return focal.mean()
\end{lstlisting}

\section{ETHOS Encoder Forward Pass}
%==============================================================================

\begin{lstlisting}[language=Python, caption=ETHOS Encoder (Simplified)]
class ETHOSEncoder(nn.Module):
    def __init__(self, d_model=128, nhead=8, num_layers=2):
        super().__init__()
        self.embed = nn.Linear(4, d_model)  # 4 = token dims
        encoder_layer = nn.TransformerEncoderLayer(
            d_model, nhead, dim_feedforward=512, dropout=0.2
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers)
        self.d_model = d_model

    def forward(self, x):
        # x: (B, L, 4)
        h = self.embed(x)  # (B, L, d_model)
        h = h.transpose(0, 1)  # (L, B, d_model)
        h = self.transformer(h)
        h = h.mean(dim=0)  # (B, d_model)
        return h
\end{lstlisting}

\section{FedAvg Server Update}
%==============================================================================

\begin{lstlisting}[language=Python, caption=FedAvg Aggregation]
def fedavg_aggregate(client_weights, client_sizes):
    total = sum(client_sizes)
    global_state = {}
    for key in client_weights[0].keys():
        global_state[key] = sum(
            w[key] * n for w, n in zip(client_weights, client_sizes)
        ) / total
    return global_state
\end{lstlisting}
