\section{Limitations}
%==============================================================================

\begin{itemize}
\item \textbf{Data}: MIMIC-IV is retrospective; prospective validation needed.
\item \textbf{Models}: ETHOS underperforms RF; negative result reported honestly.
\item \textbf{Generalization}: Results may not transfer to other hospitals or countries.
\item \textbf{Bias}: Age/sex stratification performed; full fairness audit recommended.
\end{itemize}
