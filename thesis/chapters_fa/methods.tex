\subsection{استخراج کوهورت}

یک کوهورت شامل \pnum{2,000} بیمار از \lr{MIMIC\text{-}IV v3.1} با معیارهای زیر استخراج می‌شود:
\begin{itemize}
\item فقط اولین بستری در \lr{ICU} (بدون بستری مجدد در کوهورت).
\item حداقل $4$ ساعت مدت بستری (برای اطمینان از کفایت داده).
\item سن $\geq$ $18$ سال در پذیرش.
\item تشخیص اصلی \lr{ICD-10} برای تخصیص گره بیماری موجود باشد.
\end{itemize}

فرایند استخراج (پیوست~\ref{app:sql}) ابتدا \texttt{\lr{icustays}} را فیلتر می‌کند، سپس آن را به \texttt{\lr{patients}} و \texttt{\lr{admissions}} پیوند می‌زند و در نهایت \pnum{2,000} بستری را با لایه‌بندی بر اساس گروه تشخیص اولیه نمونه‌گیری می‌کند. جدول~\ref{tab:demographics} جمعیت‌شناسی کوهورت را خلاصه می‌کند.

\subsection{فرایند تفصیلی استخراج \lr{MIMIC\text{-}IV}}

استخراج داده با یک فرایند چندمرحله‌ای \lr{SQL} انجام می‌شود که هم بازتولیدپذیر باشد و هم از نظر بالینی قابل دفاع بماند. الگوریتم~\ref{alg:extraction} این فرایند را به‌صورت رسمی نشان می‌دهد.

\begin{algorithm}[htbp]
\begin{latin}\footnotesize
\caption{فرایند استخراج کوهورت \lr{MIMIC\text{-}IV}}
\label{alg:extraction}
\begin{algorithmic}[1]
\Require MIMIC-IV database, target size $n=2000$
\State \textbf{Stage 1: Candidate ICU stays}
\State $\mathcal{C} \gets$ \texttt{SELECT} from \texttt{icustays} \texttt{WHERE} first\_stay = TRUE
\State $\mathcal{C} \gets \{c \in \mathcal{C} : \text{\lr{LOS}}(c) \geq 4\text{\lr{h}}\}$
\State \textbf{Stage 2: Age filter}
\State $\mathcal{C} \gets \{c \in \mathcal{C} : \text{\lr{age}}(c) \geq 18\}$ \Comment{Via \texttt{\lr{patients.anchor\_age}}}
\State \textbf{Stage 3: Diagnosis assignment}
\For{each $c \in \mathcal{C}$}
\State $\text{\lr{dx}}(c) \gets$ primary ICD-10 from \texttt{\lr{diagnoses\_icd}} (\texttt{\lr{seq\_num}}=1)
\State $\text{\lr{node}}(c) \gets \text{\lr{map\_to\_disease\_node}}(\text{\lr{dx}}(c))$
\EndFor
\State $\mathcal{C} \gets \{c \in \mathcal{C} : \text{\lr{node}}(c) \neq \texttt{NULL}\}$
\State \textbf{Stage 4: Stratified sampling}
\State $\mathcal{S} \gets \text{\lr{stratified\_sample}}(\mathcal{C}, n=2000, \text{\lr{strata}}=\text{\lr{node}})$
\State \textbf{Stage 5: Label derivation}
\For{each $c \in \mathcal{S}$}
\State $\text{\lr{dest}}(c) \gets$ \texttt{\lr{admissions.discharge\_location}}
\State $\text{\lr{readmit}}(c) \gets$ check 30-day readmission from \texttt{admissions}
\State $y(c) \gets \mathbb{1}[\text{\lr{dest}}(c) \in \{\text{\lr{HOME}}, \text{\lr{REHAB}}, \ldots\} \land \neg\text{\lr{readmit}}(c)]$
\EndFor
\State \Return $\{(c, y(c)) : c \in \mathcal{S}\}$
\end{algorithmic}
\end{latin}
\end{algorithm}

ساختار پیوند بر سه کلید اصلی تکیه دارد: \texttt{\lr{subject\_id}} بیماران را به پذیرش‌ها، \texttt{\lr{hadm\_id}} پذیرش‌ها را به تشخیص‌ها و سوابق ترخیص، و \texttt{\lr{stay\_id}} بستری‌های \lr{ICU} را به رویدادهای آزمایشگاهی و علائم حیاتی وصل می‌کند. ترتیب پیوندها طوری انتخاب شده است که اندازۀ نتایج میانی تا حد ممکن کاهش یابد. ابتدا بستری‌های \lr{ICU} فیلتر می‌شوند و تعداد نمونه‌ها از بیش از \pnum{250} هزار به حدود \pnum{180} هزار بستری اول می‌رسد. سپس فیلترهای سن و \lr{LOS} اعمال می‌شوند و تعداد به حدود \pnum{150} هزار کاهش می‌یابد. در گام بعد، وجود تشخیص \lr{ICD-10} الزامی می‌شود و حدود \pnum{120} هزار نمونه باقی می‌ماند. در نهایت، \pnum{2,000} بیمار به‌صورت لایه‌بندی‌شده نمونه‌گیری می‌شوند. این فیلتر آبشاری باعث می‌شود پیوندهای پرهزینه با \texttt{\lr{labevents}} و \texttt{\lr{chartevents}} فقط روی کوچک‌ترین کوهورت لازم اجرا شوند.

\subsection{استخراج آزمایشگاه و علائم حیاتی}

از \pnum{15} \lr{itemid} آزمایشگاهی و \pnum{7} \lr{itemid} علائم حیاتی از \texttt{\lr{d\_labitems}} و \texttt{\lr{d\_items}} در \lr{MIMIC\text{-}IV} استفاده می‌شود. مقادیر از \pnum{24} ساعت اول بستری \lr{ICU} استخراج می‌شوند. هر اندازه‌گیری به بازه‌ی بالینی \lr{clip} می‌شود (مثلاً کراتینین \lr{[0.1, 30] mg/dL}) تا پرت‌ها حذف شوند و سپس با \lr{min-max} به بازۀ \cnum{[0,1]} نرمال می‌شود:
\begin{equation}
x' = \frac{\text{\lr{clip}}(x, x_{\min}, x_{\max}) - x_{\min}}{x_{\max} - x_{\min}}
\end{equation}
جدول‌های~\ref{tab:lab_ranges} و~\ref{tab:vital_ranges} بازه‌های آزمایشگاهی و علائم حیاتی را فهرست می‌کنند.

\input{tables_ranges}

$15$ اندازه‌گیری آزمایشگاهی بر اساس ارتباط بالینی با پیش‌بینی ترخیص \lr{ICU} و دسترسی در \lr{MIMIC\text{-}IV} انتخاب شدند:
\begin{itemize}
\item \textbf{پانل کلیوی}: کراتینین، \lr{BUN} (نیتروژن اوره خون)، پتاسیم، سدیم، کلر، بی‌کربنات
\item \textbf{هماتولوژی}: هموگلوبین، پلاکت، شمارش گلبول سفید (\lr{WBC})
\item \textbf{متابولیک}: گلوکز، لاکتات، آلبومین
\item \textbf{کبدی}: \lr{ALT} (آلانین آمینوترانسفراز)، بیلی‌روبین
\item \textbf{انعقاد}: \lr{INR} (نسبت نرمال‌شده بین‌المللی)
\end{itemize}

$7$ علامت حیاتی عبارت‌اند از: ضربان قلب، فشار خون سیستولیک، فشار خون دیاستولیک، فشار شریانی میانگین، دما، \lr{SpO$_2$} (اشباع اکسیژن)، و نرخ تنفس. وقتی چند اندازه‌گیری در پنجره‌ی $24$-ساعته وجود دارد، از مقدار \emph{میانه} برای کاهش حساسیت به جهش‌های گذرا یا آرتیفکت اندازه‌گیری استفاده می‌شود.

\subsection{مهندسی ویژگی}

سه ویژگی مهندسی‌شده به $22$ ویژگی پایه ($15$ آزمایشگاه + $7$ علائم حیاتی) افزوده می‌شود:
\begin{enumerate}
\item \textbf{شاخص شوک}: $\text{\lr{SI}} = \text{\lr{HR}} / \text{\lr{SBP}}$، با بازه‌ی بالینی $[0.3, 2.0]$ به $[0,1]$ نرمال می‌شود. شاخص شوک نشانگر اعتباریافته‌ی ناپایداری همودینامیک است؛ مقادیری فراتر از \mbox{\lr{$1.0$}} برای \lr{SI} با افزایش مرگ‌ومیر و طولانی‌شدن بستری \lr{ICU} همراه‌اند. به‌صورت رسمی:
\begin{equation}
\text{\lr{SI}}_{\text{\lr{norm}}} = \frac{\text{\lr{clip}}(\text{\lr{HR}}/\text{\lr{SBP}}, 0.3, 2.0) - 0.3}{2.0 - 0.3}
\end{equation}

\item \textbf{جایگزین \lr{SOFA}}: تقریب ساده‌شده‌ی نمره‌ی \lr{Sequential Organ Failure Assessment} به‌صورت زیر محاسبه می‌شود:
\begin{equation}
\text{\lr{SOFA}}_{\text{\lr{proxy}}} = \frac{1}{4}\left[(1 - x_{\text{\lr{creat}}}) + (1 - x_{\text{\lr{plt}}}) + (1 - x_{\text{\lr{SpO}}_2}) + x_{\text{\lr{lactate}}}\right]
\end{equation}
که در آن هر $x$ مقدار نرمال‌شده‌ی \lr{min-max} است. مقادیر بالاتر جایگزین نشان‌دهنده‌ی عملکرد اندام بدتر است. این تقریب چهار از شش حوزه‌ی \lr{SOFA} (کلیوی، انعقادی، تنفسی، قلبی‌عروقی) را با داده‌ی آزمایشگاهی و علائم حیاتی موجود می‌گیرد. حوزه‌های عصبی (\lr{GCS}) و کبدی (بیلی‌روبین) به‌دلیل فقدان داده حذف شده‌اند؛ بیلی‌روبین در دسترس است اما مستقل به مجموعه‌ی ویژگی خام کمک می‌کند.

\item \textbf{شمارش آزمایش‌های غیرطبیعی}: کسر مقادیر آزمایشگاهی با انحراف شدید:
\begin{equation}
\text{\lr{AbnCount}} = \frac{1}{15}\sum_{j=1}^{15} \mathbb{1}[|x_j - 0.5| > 0.35]
\end{equation}
این تعداد آزمایش‌های خارج از $30\%$ میانی بازه‌ی نرمال‌شده‌ی خود را می‌گیرد و انحراف کلی فیزیولوژیک را منعکس می‌کند. بیماری با $8$ از $15$ آزمایش غیرطبیعی از نظر بالینی بیمارتر از بیماری با $2$ مورد غیرطبیعی است، صرف‌نظر از اینکه کدام آزمایش‌ها منحرف شده‌اند.
\end{enumerate}
مجموعه‌ی ویژگی جدولی نهایی $25$ بعد دارد. جدول~\ref{tab:feature_summary} همه‌ی ویژگی‌ها را دسته‌بندی می‌کند.

\subsection{راهبرد داده‌ی گمشده}

داده‌های بالینی ذاتاً ناقص‌اند. راهبرد جایگزینی ما سلسله‌مراتب محافظه‌کارانه‌ای را دنبال می‌کند:
\begin{enumerate}
\item \textbf{پرکردن رو به جلو درون‌بیماری}: اگر مقدار آزمایش در ساعت $2$ و $18$ اندازه گرفته شده باشد، پرس‌وجو در ساعت $12$ از مقدار ساعت $2$ استفاده می‌کند (بدون پرکردن معکوس برای جلوگیری از نشت اطلاعات).
\item \textbf{تجمیع میانه‌ای}: برای خلاصه‌ی $24$ ساعت اول، میانه‌ی همه‌ی اندازه‌گیری‌های موجود در پنجره گرفته می‌شود. میانه نسبت به پرت‌های ناشی از آرتیفکت اندازه‌گیری مقاوم است.
\item \textbf{جایگزینی با میانه‌ی جمعیت}: اگر در $24$ ساعت اندازه‌گیری برای ویژگی‌ای موجود نباشد، میانه‌ی جمعیت (فقط روی مجموعه‌ی آموزش محاسبه‌شده برای جلوگیری از نشت داده) جایگزین می‌شود.
\item \textbf{شاخص فقدان}: در بردار ویژگی $25$-بعدی شاخص صریح فقدان استفاده نمی‌شود، اما ویژگی شمار آزمایش غیرطبیعی به‌صورت ضمنی اطلاعاتی درباره‌ی ویژگی‌هایی با مقادیر شدیداً منحرف می‌گیرد.
\end{enumerate}

نرخ فقدان در کوهورت ما: آزمایش‌های روتین (کراتینین، سدیم، پتاسیم) $5$-$12\%$؛ آزمایش‌های تخصصی (لاکتات، آلبومین، \lr{INR}) $18$-$35\%$؛ علائم حیاتی $2$-$8\%$. ویژگی‌هایی با بیش از $50\%$ فقدان در مرحله‌ی انتخاب ویژگی حذف شدند؛ همه‌ی $22$ ویژگی نگه‌داشته‌شده حداکثر $35\%$ فقدان دارند.

\subsection{تقسیم داده}

از تقسیم‌های ثابت آموزش/اعتبارسنجی/آزمون با بذر $42$ استفاده می‌شود:
\begin{itemize}
\item $80\%$ آموزش+اعتبارسنجی، $20\%$ آزمون (لایه‌بندی‌شده بر اساس برچسب).
\item از $80\%$، $80\%$ آموزش و $20\%$ اعتبارسنجی.
\item نهایی: $64\%$ آموزش (\pnum{1,280} بیمار)، $16\%$ اعتبارسنجی (\pnum{320} بیمار)، $20\%$ آزمون (\pnum{400} بیمار).
\end{itemize}
هیچ داده‌ی آزمونی برای انتخاب مدل یا تنظیم ابرپارامتر استفاده نمی‌شود. همه‌ی تصمیم‌ها روی مجموعه‌ی اعتبارسنجی اعتبار می‌یابند و روی مجموعه‌ی آزمون گزارش می‌شوند. لایه‌بندی تضمین می‌کند نرخ امکان‌پذیری (تقریباً $42\%$ قابل‌تحقق) در همه‌ی تقسیم‌ها حفظ شود. برای آزمایش‌های فدرال، مجموعه‌ی آموزش بیشتر بر اساس گره بیماری افراز می‌شود و هر گره سهم متناسب از داده‌ی آموزش دریافت می‌کند.

\subsection{ساختار توکن}

هر بیمار به‌صورت دنباله‌ای حداکثر $25$ توکن بازنمایی می‌شود. هر توکن چهار جزء دارد:
\begin{equation}
\text{توکن} = [\text{\lr{symptom\_id}}, \text{\lr{intensity}}, \text{\lr{time\_delta}}, \text{\lr{modality}}]
\end{equation}
که در آن:
\begin{itemize}
\item \textbf{\lr{symptom\_id}}: $1$-$15$ برای آزمایشگاه‌ها، $16$-$22$ برای علائم حیاتی، $0$ برای \lr{MASK} (پدینگ).
\item \textbf{\lr{intensity}}: مقدار نرمال‌شده در $[0,1]$.
\item \textbf{\lr{time\_delta}}: جابجایی زمانی نرمال‌شده ($0$ برای تجمیع $24$ ساعت اول).
\item \textbf{\lr{modality}}: $0$ برای آزمایشگاه، $1$ برای علائم حیاتی.
\end{itemize}

\subsection{ساخت دنباله}

برای هر بیمار، همه‌ی اندازه‌گیری‌های غیرگمشده جمع‌آوری، بر اساس \lr{symptom\_id} مرتب، و با توکن‌های \lr{MASK} به طول $25$ پد می‌شوند. این دنباله‌ی با طول ثابت برای رمزگذارهای \lr{transformer} مناسب است.

\subsection{ارتباط با ویژگی‌های جدولی}

بازنمایی جدولی ($25$ ویژگی) نسخه‌ی مسطح و تجمیع‌شده‌ی دنباله‌ی توکن است. هر دو از همان مقادیر نرمال‌شده‌ی زیرین استفاده می‌کنند. مدل‌های جدولی بردار $25$-بعدی را دریافت می‌کنند؛ \lr{ETHOS} دنباله‌ی توکن را دریافت می‌کند.

\section{معماری رمزگذار \lr{ETHOS}}
%==============================================================================

\subsection{نمای کلی}

\lr{ETHOS}~\cite{sharafi2023ethos} یک رمزگذار \lr{transformer} است که دنباله‌های توکن را به بردارهای نهفتۀ با اندازه ثابت نگاشت می‌کند. شامل موارد زیر است:
\begin{itemize}
\item لایۀ نهفته‌سازی: \lr{(symptom\_id, intensity, time\_delta, modality)} را به بردارهای $d$-بعدی نگاشت می‌کند.
\item کدگذاری موقعیتی: سینوسی یا آموخته‌شده.
\item $L$ لایه‌ی \lr{transformer}: خودتوجه چندسر + پیش‌خوران.
\item تجمیع: میانگین روی دنباله (به‌جز \lr{MASK}) یا توکن به‌سبک \lr{[CLS]}.
\end{itemize}

\subsection{پیکربندی}

از \texttt{\lr{config.yaml}}:
\begin{itemize}
\item $d_{\text{\lr{model}}} = 128$
\item $n_{\text{\lr{head}}} = 8$
\item $n_{\text{\lr{layers}}} = 2$
\item $d_{\text{\lr{ff}}} = 512$
\item $\text{\lr{Dropout}} = 0.2$
\item $\text{\lr{vocab\_size}} = 23$ ($22$ علامت + $1$ \lr{MASK})
\end{itemize}

\subsection{خروجی}

رمزگذار برای هر بیمار یک نهفته‌سازی $128$-بعدی تولید می‌کند. این نهفته‌سازی توسط سر نمونه‌ی اولیه‌ای برای طبقه‌بندی کم‌نمونه استفاده می‌شود.

\subsection{جزئیات مکانیزم توجه}

خودتوجه چندسر به‌صورت زیر محاسبه می‌شود:
\begin{equation}
\text{\lr{Attention}}(Q, K, V) = \text{\lr{softmax}}\left(\frac{QK^\top}{\sqrt{d_k}}\right) V
\end{equation}
که در آن $Q$، $K$، $V$ تصویرهای خطی ورودی‌اند. $8$ سر با $d_k = d_{\text{\lr{model}}}/8 = 16$ استفاده می‌شود. لایه‌ی پیش‌خوران یک \lr{MLP} دولایه با \lr{ReLU} و بعد $512$ اعمال می‌کند. اتصالات باقیمانده و نرمال‌سازی لایه آموزش را پایدار می‌کنند.

%==============================================================================
\section{پیش‌بینی درمان کم‌نمونه}
%==============================================================================

\subsection{پروتکل اپیزودیک}

هر اپیزود نمونه‌برداری می‌کند:
\begin{itemize}
\item $K$ نمونه پشتیبان به‌ازای هر کلاس (پیش‌فرض $K=5$).
\item $Q$ نمونه پرس‌وجو به‌ازای هر کلاس (پیش‌فرض $Q=15$).
\item $2$ کلاس: قابل‌تحقق ($1$) و غیرقابل‌تحقق ($0$).
\end{itemize}
جمع: $2K$ پشتیبان، $2Q$ پرس‌وجو در هر اپیزود.

\subsection{سر نمونه‌ی اولیه‌ای}

برای مجموعه پشتیبان $\mathcal{S}_k$ کلاس $k$:
\begin{equation}
\mathbf{c}_k = \frac{1}{K} \sum_{(x_i, y_i) \in \mathcal{S}_k} f_\theta(x_i)
\end{equation}
برای پرس‌وجو $x$، لگاریتم‌های شانس:
\begin{equation}
z_k = \frac{\langle f_\theta(x), \mathbf{c}_k \rangle}{\|f_\theta(x)\| \|\mathbf{c}_k\|} \cdot \tau
\end{equation}
که در آن $\tau = 10$ دماست. احتمالات: $\text{\lr{softmax}}(z)$.

\subsection{تابع هزینه}

از زیان کانونی برای مدیریت نامتعادلی کلاس استفاده می‌شود:
\begin{equation}
\mathcal{L} = -\alpha (1-p)^\gamma \log p
\end{equation}
با $\gamma = 2$ و $\alpha$ از وزن‌های کلاس. آنتروپی متقاطع حالت ویژه‌ی $\gamma = 0$ است.

\subsection{تصویر فضای متریک}

برای بهبود توان تفکیک فضای نهفته، یک \lr{MLP} تصویر اختیاری بین خروجی رمزگذار و محاسبه‌ی نمونه‌ی اولیه‌ای بررسی می‌شود:
\begin{equation}
g_\phi(\mathbf{h}) = W_2 \cdot \text{\lr{GELU}}(\text{\lr{BN}}(W_1 \cdot \mathbf{h} + b_1)) + b_2
\end{equation}
که در آن $W_1 \in \mathbb{R}^{d_h \times d}$، $W_2 \in \mathbb{R}^{d_p \times d_h}$، با $d = 128$، $d_h = 256$، $d_p = 64$. \lr{BatchNorm} و \lr{Dropout} ($0.2$) پس از لایه‌ی اول اعمال می‌شوند. تصویر بعد نهفته را از $128$ به $64$ کاهش می‌دهد و فضای متریک فشرده‌ای برای محاسبه‌ی فاصله‌ی نمونه‌ی اولیه‌ای بهینه می‌سازد. پارامتر دما $\tau$ با $\tau = \exp(\log \tau_0)$ قابل‌آموختن می‌شود تا مدل تیزی توزیع احتمال را تطبیقی کنترل کند.

\subsection{آموزش}

تا \pnum{1,500} اپیزود آموزش، صبر زودتوقف $10$. بهینه‌ساز: \lr{AdamW} با وزن‌زدایی $10^{-4}$ و زمان‌بندی نرخ یادگیری کسینوسی (از $10^{-3}$ تا $10^{-5}$). رمزگذار و سر نرخ‌های یادگیری متفاوت دارند (رمزگذار: $10^{-4}$، سر: $10^{-3}$) تا بازنمایی‌های ازپیش‌آموزش‌دیده حفظ شود درحالی‌که سر طبقه‌بندی تطبیق می‌یابد. برش گرادیان (حداکثر نرم $1.0$) ناپایدگی آموزش را مهار می‌کند. اعتبارسنجی: $100$ اپیزود.

%==============================================================================
\section{سر شبکه‌ی رابطه (جایگزین)}
%==============================================================================

به‌عنوان جایگزین سر نمونه‌ی اولیه‌ای، سر شبکه‌ی رابطه~\cite{sung2018learning} پیاده‌سازی می‌شود که شباهت کسینوسی ثابت را با ماژول رابطه‌ی آموخته‌شده جایگزین می‌کند. با بردار نهفتۀ پرس‌وجو $\mathbf{q} = f_\theta(x)$ و نمونه‌ی اولیه‌ی کلاس $\mathbf{c}_k$، نمره‌ی رابطه توسط \lr{MLP} دولایه محاسبه می‌شود:

\begin{equation}
\mathbf{h} = \text{\lr{ReLU}}(W_1 [\mathbf{q}; \mathbf{c}_k; \mathbf{q} \odot \mathbf{c}_k; |\mathbf{q} - \mathbf{c}_k|] + b_1) \in \mathbb{R}^{256}
\end{equation}
\begin{equation}
r_k = \sigma(W_2 \mathbf{h} + b_2) \in [0,1]
\end{equation}

که در آن ورودی چهار ویژگی تعامل را به‌هم می‌چسباند: بردارهای نهفتۀ خام ($2 \times 128 = 256$ بعد)، حاصل‌ضرب درایه‌ای ($128$ بعد)، و قدر مطلق تفاوت ($128$ بعد)، که ورودی $640$-بعدی به ماژول رابطه می‌دهد. هزینه میانگین مربعات خطاست:
\begin{equation}
\mathcal{L}_{\text{\lr{relation}}} = \frac{1}{|\mathcal{Q}|} \sum_{(x,y) \in \mathcal{Q}} \sum_{k=0}^{1} (r_k(x) - \mathbb{1}[y=k])^2
\end{equation}

الگوریتم~\ref{alg:relation} رویه‌ی آموزش شبکه‌ی رابطه‌ی اپیزودیک را شرح می‌دهد.

\begin{algorithm}[htbp]
\begin{latin}\footnotesize
\caption{آموزش اپیزودیک شبکۀ رابطه}
\label{alg:relation}
\begin{algorithmic}[1]
\Require Encoder $f_\theta$, relation module $g_\phi$, support shots $K$, query shots $Q$
\For{each episode}
\State Sample $\mathcal{S}_0, \mathcal{S}_1$ ($K$ per class), $\mathcal{Q}_0, \mathcal{Q}_1$ ($Q$ per class)
\State $\mathbf{c}_k \gets \frac{1}{K}\sum_{x \in \mathcal{S}_k} f_\theta(x)$ for $k \in \{0,1\}$
\State $\mathcal{L} \gets 0$
\For{each $(x, y) \in \mathcal{Q}_0 \cup \mathcal{Q}_1$}
\State $\mathbf{q} \gets f_\theta(x)$
\For{$k \in \{0, 1\}$}
\State $r_k \gets g_\phi(\mathbf{q}, \mathbf{c}_k)$
\State $\mathcal{L} \gets \mathcal{L} + (r_k - \mathbb{1}[y=k])^2$
\EndFor
\EndFor
\State Update $\theta, \phi$ via backpropagation on $\mathcal{L}$
\EndFor
\State \Return $\theta, \phi$
\end{algorithmic}
\end{latin}
\end{algorithm}

شبکه‌ی رابطه تقریباً $0.2$ مگاپارامتر، برای \lr{MLP} رابطه، به $1.8$ مگاپارامتر رمزگذار می‌افزاید. در آزمایش حذف‌بخشی، سر رابطه \lr{AUROC} قابل مقایسه‌ای با سر نمونه‌ی اولیه‌ای به‌دست آورد؛ اختلاف در حدود $0.01$ بود. این نتیجه نشان می‌دهد برای این وظیفۀ دودویی، متریک کسینوسی ثابت تا حد زیادی کافی است و گلوگاه اصلی بیشتر در بازنمایی‌های رمزگذار قرار دارد، نه در تابع شباهت.

\subsection{پروتکل \lr{FedAvg}}

پروتکل استاندارد میانگین‌گیری فدرال (\lr{FedAvg})~\cite{mcmahan2017communication} با پیکربندی زیر پیاده‌سازی می‌شود:
\begin{itemize}
\item \textbf{دورهای ارتباطی}: $T = 20$. \lr{AUROC} اعتبارسنجی تا دور $15$-$20$ اشباع می‌شود.
\item \textbf{دوره‌های محلی}: $E = 3$ در هر دور. هر گره پیش از ارسال به‌روزرسانی‌ها $3$ گذر کامل روی داده‌ی محلی آموزش می‌بیند.
\item \textbf{اندازه‌ی دسته‌ی محلی}: $32$.
\item \textbf{وزن‌های تجمیع}: متناسب با اندازه‌ی مجموعه‌داده‌ی محلی: $w_k = n_k / \sum_j n_j$.
\item \textbf{نرخ یادگیری}: همانند متمرکز ($10^{-3}$ برای سر، $10^{-4}$ برای رمزگذار).
\item \textbf{انتخاب کلاینت}: مشارکت کامل (همه‌ی $5$ گره در هر دور شرکت می‌کنند).
\end{itemize}

گام تجمیع سرور میانگین وزنی همه‌ی پارامترهای مدل را محاسبه می‌کند:
\begin{equation}
\theta_{t+1} = \sum_{k=1}^{5} \frac{n_k}{\sum_j n_j} \theta_{t}^{(k)}
\end{equation}

برای معماری ما، مدل کامل ($\approx$$2$ مگاپارامتر) هر دور ارسال می‌شود. هزینه‌ی ارتباطی کل: \LTRformula{5\,\mathrm{nodes} \times 20\,\mathrm{rounds} \times 2\,\mathrm{directions} \times 8\,\mathrm{MB} \approx 1.6\,\mathrm{GB}} است.

\subsection{پارامترهای مشترک در برابر محلی}

رمزگذار \lr{ETHOS} بین گره‌ها مشترک است و همه‌ی گره‌ها از بازنمایی سراسری بهره می‌برند. سر نمونه‌ی اولیه‌ای نیز مشترک است؛ سرهای محلی به‌دلیل ناکافی بودن داده‌ی محلی برای تخصص سر، عملکرد را $1$-$2\%$ \lr{AUROC} کاهش می‌دهند. هر گره گرادیان روی داده‌ی محلی محاسبه و مدل به‌روزشده‌ی کامل را به سرور می‌فرستد.

\subsection{پایش همگرایی}

همگرایی با ردیابی \lr{AUROC} اعتبارسنجی سراسری پس از هر دور پایش می‌شود. معیار همگرایی: بدون بهبود در \lr{AUROC} اعتبارسنجی برای $5$ دور متوالی. در عمل، همگرایی برای بیشتر پیکربندی‌های ابرپارامتر تا دور $15$-$18$ رخ می‌دهد. $T=20$ به‌عنوان کران بالای محافظه‌کارانه تنظیم شده است.

%==============================================================================
\section{معماری ترکیبی \lr{ETHOS} و جنگل تصادفی}
\label{sec:hybrid}
%==============================================================================

با انگیزه از مشاهده‌ی اینکه بردارهای نهفتۀ \lr{ETHOS} سیگنال مکمل نسبت به ویژگی‌های جدولی مهندسی‌شده می‌گیرند، معماری ترکیبی طراحی می‌شود. برای هر بیمار $i$:
\begin{equation}
\mathbf{x}_{\text{\lr{hybrid}}}^{(i)} = [\mathbf{e}_{\text{\lr{ETHOS}}}^{(i)}; \mathbf{x}_{\text{\lr{tab}}}^{(i)}] \in \mathbb{R}^{153}
\end{equation}
که در آن $\mathbf{e}_{\text{\lr{ETHOS}}} \in \mathbb{R}^{128}$ نهفته‌سازی \lr{CLS} \lr{ETHOS} و $\mathbf{x}_{\text{\lr{tab}}} \in \mathbb{R}^{25}$ بردار ویژگی جدولی مهندسی‌شده است. طبقه‌بند \lr{Random Forest} با \lr{RandomizedSearchCV} ($50$ آزمایش، اعتبارسنجی متقاطع لایه‌بندی‌شده $5$-تایی) روی بازنمایی به‌هم‌چسبانده‌شده آموزش داده می‌شود. تنظیم آستانه روی مجموعه‌ی اعتبارسنجی \lr{F1} را بیشینه می‌کند. این معماری ظرفیت یادگیری بازنمایی \lr{transformer} را با طبقه‌بندی مقاوم روش‌های گروهی مبتنی‌بر گرادیان ترکیب می‌کند و گلوگاه بازنمایی شناسایی‌شده در مطالعه‌ی حذف‌بخشی را پوشش می‌دهد.

همچنین \lr{RF} تنها روی بردارهای نهفتۀ \lr{ETHOS} ($128$-بعدی) ارزیابی می‌شود تا سهم بازنمایی‌های آموزش‌دیده از ویژگی‌های مهندسی‌شده جدا شود.

\section{پروتکل ارزیابی}
%==============================================================================

\subsection{معیارها}

چند معیار برای پوشش جنبه‌های مختلف عملکرد گزارش می‌شود. هر معیار رسماً در ادامه تعریف می‌شود.

\subsubsection{\lr{AUROC} (مساحت زیر منحنی مشخصه‌ی عملیاتی گیرنده)}

معیار ارزیابی اصلی. برای طبقه‌بند دودویی با تابع نمره $s(x)$، منحنی \lr{ROC} نرخ مثبت درست (\lr{TPR}، حساسیت) را در برابر نرخ مثبت نادرست (\lr{FPR}، $1$ منهای اختصاصیت) با تغییر آستانه‌ی تصمیم رسم می‌کند:
\begin{equation}
\text{\lr{AUROC}} = \int_0^1 \text{\lr{TPR}}(t) \, d(\text{\lr{FPR}}(t)) = P(s(x^+) > s(x^-))
\end{equation}
که در آن $x^+$ نمونه‌ی مثبت و $x^-$ نمونه‌ی منفی به‌صورت تصادفی انتخاب‌شده‌اند. \lr{AUROC} نسبت به آستانه نامتغیر و نسبت به نامتعادلی کلاس مقاوم است. \lr{AUROC} با قاعده‌ی ذوزنقه‌ای (\texttt{\lr{sklearn.metrics.roc\_auc\_score}}) محاسبه می‌شود. مقادیر: $0.5$ تصادفی؛ $0.7$-$0.8$ قابل‌قبول برای پشتیبانی تصمیم بالینی؛ $0.8$-$0.9$ خوب؛ ارقامی بالای \mbox{\lr{$0.9$}} عالی. بهترین مدل ما (\lr{AUROC} $0.76$) در بازه‌ی قابل‌قبول قرار دارد.

\subsubsection{دقت، فراخوانی، و امتیاز \lr{F1}}

برای کلاس $k$ با مثبت درست (\lr{TP})، مثبت نادرست (\lr{FP})، و منفی نادرست (\lr{FN}):
\begin{align}
\text{\lr{Precision}}_k &= \frac{\text{\lr{TP}}_k}{\text{\lr{TP}}_k + \text{\lr{FP}}_k} \\
\text{\lr{Recall}}_k &= \frac{\text{\lr{TP}}_k}{\text{\lr{TP}}_k + \text{\lr{FN}}_k} \\
\text{\lr{F1}}_k &= \frac{2 \cdot \text{\lr{Precision}}_k \cdot \text{\lr{Recall}}_k}{\text{\lr{Precision}}_k + \text{\lr{Recall}}_k} = \frac{2\text{\lr{TP}}_k}{2\text{\lr{TP}}_k + \text{\lr{FP}}_k + \text{\lr{FN}}_k}
\end{align}

از \textbf{امتیاز \lr{F1} کلان}، یعنی میانگین بدون وزن امتیازهای \lr{F1} به‌ازای هر کلاس، برای جلوگیری از تسلط کلاس اکثریت استفاده می‌شود:
\begin{equation}
\text{\lr{Macro-F1}} = \frac{1}{N} \sum_{k=1}^{N} \text{\lr{F1}}_k = \frac{\text{\lr{F1}}_0 + \text{\lr{F1}}_1}{2}
\end{equation}

\subsubsection{ضریب همبستگی متیوز (\lr{MCC})}

\lr{MCC} معیاری متعادل برای طبقه‌بندی دودویی فراهم می‌کند و هر چهار خانه‌ی ماتریس سردرگمی را در نظر می‌گیرد:
\begin{equation}
\text{\lr{MCC}} = \frac{\text{\lr{TP}} \cdot \text{\lr{TN}} - \text{\lr{FP}} \cdot \text{\lr{FN}}}{\sqrt{(\text{\lr{TP}}+\text{\lr{FP}})(\text{\lr{TP}}+\text{\lr{FN}})(\text{\lr{TN}}+\text{\lr{FP}})(\text{\lr{TN}}+\text{\lr{FN}})}}
\end{equation}
\lr{MCC} از $-1$ (طبقه‌بندی کاملاً اشتباه) از طریق $0$ (پیش‌بینی تصادفی) تا $+1$ (طبقه‌بندی کامل) متغیر است. برخلاف دقت، \lr{MCC} حتی با کلاس‌های بسیار نامتعادل معنادار است. مدلی که همیشه کلاس اکثریت را پیش‌بینی کند دقت $\approx 0.58$ ولی \lr{MCC} $= 0$ دارد.

\subsubsection{دقت (\lr{Accuracy})}

دقت طبقه‌بندی استاندارد:
\begin{equation}
\text{\lr{Accuracy}} = \frac{\text{\lr{TP}} + \text{\lr{TN}}}{\text{\lr{TP}} + \text{\lr{TN}} + \text{\lr{FP}} + \text{\lr{FN}}}
\end{equation}
دقت برای کامل بودن گزارش می‌شود اما به‌عنوان معیار اصلی به‌دلیل نامتعادلی کلاس ($42\%$ قابل‌تحقق) استفاده نمی‌شود.

\subsubsection{نمره‌ی \lr{Brier}}

برای ارزیابی کالیبراسیون احتمالی:
\begin{equation}
\text{\lr{Brier}} = \frac{1}{n} \sum_{i=1}^{n} (p_i - y_i)^2
\end{equation}
که در آن $p_i$ احتمال پیش‌بینی‌شده و $y_i \in \{0,1\}$ برچسب درست است. کمتر بهتر است؛ $0.25$ متناظر با پیش‌بیننده‌ی ثابت در $0.5$ است.

\subsubsection{معناداری و محدودیت مقایسۀ آماری}

معناداری با رویکردهای زیر و با احتیاط تفسیر می‌شود:
\begin{enumerate}
\item \textbf{فواصل اطمینان \lr{bootstrap}}: مجموعه‌ی آزمون با جایگزینی \pnum{1,000} بار نمونه‌برداری مجدد، معیار برای هر نمونه محاسبه، و صدک‌های $2.5$-$97.5$ به‌عنوان فاصله اطمینان $95\%$ گزارش می‌شود.
\item \textbf{آزمون \lr{DeLong}}~\cite{delong1988comparing}: فقط زمانی معتبر است که دو مدل روی همان نمونه‌های آزمون و با بردارهای پیش‌بینی هم‌تراز ارزیابی شده باشند. فرض صفر برابری دو \lr{AUROC} است.
\end{enumerate}
از آنجا که همۀ مدل‌های این پایان‌نامه بردارهای پیش‌بینی هم‌تراز تحت یک پروتکل واحد تولید نمی‌کنند، مقایسۀ جفتی رسمی برای همۀ سطرهای نتایج قابل ادعا نیست. بنابراین در فصل نتایج، بازه‌های اطمینان در مواردی که واقعاً تولید شده‌اند گزارش می‌شود و مقایسه‌های بین‌پروتکلی به‌صورت توصیفی تفسیر می‌گردد.

\subsubsection{پایداری (واریانس)}

برای ارزیابی اپیزودیک کم‌نمونه، انحراف معیار \lr{AUROC} در $100$ اپیزود نمونه‌برداری‌شده گزارش می‌شود:
\begin{equation}
\text{\lr{Stability}} = \sqrt{\frac{1}{M-1}\sum_{m=1}^{M}(\text{\lr{AUROC}}_m - \overline{\text{\lr{AUROC}}})^2}
\end{equation}
که در آن $M = 100$ اپیزود است. واریانس کمتر عملکرد کم‌نمونه پایدارتر در ترکیب‌های تصادفی اپیزود را نشان می‌دهد.

\subsection{آموزش کم‌نمونه نمونه‌ی اولیه‌ای}

الگوریتم~\ref{alg:proto} رویه‌ی آموزش اپیزودیک را خلاصه می‌کند. هر اپیزود مجموعه پشتیبان $\mathcal{S}$ و مجموعه پرس‌وجو $\mathcal{Q}$ را نمونه‌برداری می‌کند؛ مدل نمونه‌های اولیه را از $\mathcal{S}$ محاسبه و آنتروپی متقاطع روی $\mathcal{Q}$ را کمینه می‌کند.

\begin{algorithm}[htbp]
\begin{latin}\footnotesize
\caption{آموزش اپیزودیک نمونه‌اولیه‌ای}
\label{alg:proto}
\begin{algorithmic}[1]
\Require Encoder $f_\theta$, support shots $K$, query shots $Q$, temperature $\tau$
\For{each episode}
\State $\mathcal{L} \gets 0$
\State Sample $\mathcal{S}_0, \mathcal{S}_1$ ($K$ examples per class)
\State Sample $\mathcal{Q}_0, \mathcal{Q}_1$ ($Q$ examples per class)
\State $\mathbf{c}_k \gets \frac{1}{K}\sum_{x \in \mathcal{S}_k} f_\theta(x)$ for $k \in \{0,1\}$
\For{each $x \in \mathcal{Q}_0 \cup \mathcal{Q}_1$}
\State $z_k \gets \langle f_\theta(x), \mathbf{c}_k \rangle / (\|f_\theta(x)\| \|\mathbf{c}_k\|) \cdot \tau$
\State $\mathcal{L} \gets \mathcal{L} + \text{\lr{CrossEntropy}}(\text{\lr{softmax}}(z), y)$
\EndFor
\State Update $\theta$ via backpropagation
\EndFor
\State \Return $\theta$
\end{algorithmic}
\end{latin}
\end{algorithm}

\subsection{میانگین‌گیری فدرال}

الگوریتم~\ref{alg:fedavg} پروتکل \lr{FedAvg} را شرح می‌دهد. هر گره به‌مدت $E$ دوره به‌صورت محلی آموزش می‌بیند، سپس سرور پارامترهای مدل را با وزن اندازه‌ی مجموعه‌داده‌ی محلی تجمیع می‌کند.

\begin{algorithm}[htbp]
\begin{latin}\footnotesize
\caption{میانگین‌گیری فدرال (\lr{FedAvg})}
\label{alg:fedavg}
\begin{algorithmic}[1]
\Require Nodes $\{1,\ldots,N\}$, local data $\mathcal{D}_k$, rounds $T$
\State Initialize global model $\theta_0$
\For{round $t = 1$ to $T$}
\State Server broadcasts $\theta_t$ to all nodes
\For{each node $k$}
\State $\theta_t^{(k)} \gets \theta_t$
\For{epoch $e = 1$ to $E$}
\State $\theta_t^{(k)} \gets \theta_t^{(k)} - \eta \nabla \mathcal{L}_k(\theta_t^{(k)})$
\EndFor
\State Send $\theta_t^{(k)}$ to server
\EndFor
\State $\theta_{t+1} \gets \sum_k \frac{n_k}{n} \theta_t^{(k)}$
\EndFor
\State \Return $\theta_T$
\end{algorithmic}
\end{latin}
\end{algorithm}

\subsection{یادداشت درباره‌ی محیط الگوریتم}

شبه‌کد به‌صورت انگلیسی و با ارقام لاتین (محیط \texttt{\lr{latin}}) حروف‌چینی می‌شود تا با متن فارسی اطراف تداخل جهت و شکل ارقام نداشته باشد. از محیط‌های \texttt{\lr{algorithm}} و \texttt{\lr{algorithmic}} استفاده شده است.

\subsection{رمزگذار \lr{ETHOS}}

برای طول دنباله $L=25$، بعد نهفته $d=128$، و $L_{\text{\lr{layers}}}=2$ لایه‌ی \lr{transformer}، رمزگذار در هر گذر رو به جلو پیچیدگی $\mathcal{O}(L^2 d + L d^2)$ دارد. با اندازه‌ی دسته $32$ و $500$ اپیزود در هر دوره، آموزش تقریباً $2$-$3$ دقیقه در هر دوره روی \lr{CPU} استاندارد طول می‌کشد. شتاب‌دهی \lr{GPU} این زمان را به کمتر از $30$ ثانیه در هر دوره کاهش می‌دهد.

\subsection{ارتباط فدرال}

با $5$ گره و $20$ دور، $100$ گام ارتباطی انجام می‌شود. هر گام مدل کامل (رمزگذار + سر) را منتقل می‌کند؛ برای معماری ما این $\approx$$2$ مگاپارامتر است. در استقرار واقعی، فشرده‌سازی گرادیان یا به‌روزرسانی‌های جزئی می‌تواند پهنای‌باند را کاهش دهد.

\subsection{جایگزین‌های توکن‌سازی}

طرح‌های توکن‌سازی جایگزین در نظر گرفته شدند: (\pnum{1}) \textbf{بردارهای نهفتۀ پیوسته}: استفاده از مقادیر نرمال‌شده‌ی خام به‌عنوان نهفته‌سازی توکن پیوسته به‌جای بازه‌های گسسته؛ از شناسه‌های گسسته برای سازگاری با واژگان استاندارد \lr{transformer} استفاده می‌شود. (\pnum{2}) \textbf{توکن‌های سلسله‌مراتبی}: کدگذاری دسته آزمایش (مثلاً کلیوی، قلبی) به‌عنوان توکن والد؛ ساختار مسطح برای سادگی به‌کار می‌رود. (\pnum{3}) \textbf{بازه‌ی زمانی}: تجمیع در پنجره‌های $6$ یا $12$ ساعته؛ تجمیع $24$ ساعت اول استفاده می‌شود. کار آینده می‌تواند این جایگزین‌ها را بررسی کند.

\subsection{خط پایه‌های کلاسیک}

\lr{Random Forest} با $200$ درخت و عمق $28$ در کمتر از $1$ دقیقه روی کوهورت \pnum{2,000} بیمار آموزش می‌بیند. \lr{XGBoost} و \lr{LightGBM} نیز زمان مشابهی دارند. جستجوی ابرپارامتر، با \pnum{50} پیکربندی و اعتبارسنجی متقاطع \pnum{5}دوره برای هر مدل، حدود \pnum{10} تا \pnum{20} دقیقه به زمان اجرا می‌افزاید. کل فرایند \lr{SOTA} جدولی در کمتر از $2$ ساعت کامل می‌شود.

\section{استنتاج‌های ریاضی}
%==============================================================================

\subsection{گرادیان هزینه‌ی نمونه‌ی اولیه‌ای}

برای شبکه‌های نمونه‌ی اولیه‌ای با هزینه‌ی آنتروپی متقاطع، گرادیان هزینه نسبت به بردار نهفتۀ پرس‌وجو $f_\theta(x)$ به‌صورت زیر استنتاج می‌شود. با $p_k = \exp(z_k) / \sum_{k'} \exp(z_{k'})$ که در آن $z_k = \langle f_\theta(x), \mathbf{c}_k \rangle / \tau$. برای کلاس درست $y$:
\begin{equation}
\frac{\partial \mathcal{L}}{\partial f_\theta(x)} = \frac{1}{\tau} \sum_k (p_k - \mathbb{1}[y=k]) \mathbf{c}_k
\end{equation}
گرادیان بردار نهفتۀ پرس‌وجو را به سمت نمونه‌ی اولیه‌ی درست و دور از نمونه‌ی اولیه‌ی نادرست هل می‌دهد، با بزرگی متناسب با دما $\tau$.

\subsection{زیان کانونی}

هزینه‌ی کانونی~\cite{lin2017focal} برابر است با $\mathcal{L}_{\text{\lr{focal}}} = -\alpha (1-p_t)^\gamma \log(p_t)$ که در آن $p_t$ احتمال پیش‌بینی‌شده برای کلاس درست است. برای $\gamma = 0$ این به آنتروپی متقاطع کاهش می‌یابد. برای $\gamma > 0$، نمونه‌های آسان ($p_t$ بالا) با $(1-p_t)^\gamma$ کم‌وزن می‌شوند و هزینه روی نمونه‌های سخت متمرکز می‌شود. از $\gamma = 2$ و $\alpha$ از وزن‌های کلاس برای نامتعادلی $42/58$ استفاده می‌شود.

\subsection{فواصل اطمینان \lr{bootstrap}}

برای فواصل اطمینان \lr{bootstrap}، مجموعه‌ی آزمون با جایگزینی \pnum{1,000} بار نمونه‌برداری مجدد می‌شود، \lr{AUROC} برای هر نمونه محاسبه، و صدک‌های $2.5$ و $97.5$ به‌عنوان فاصله اطمینان $95\%$ گرفته می‌شود. مجموعه‌ی آزمون $400$ نمونه فواصل به عرض تقریبی \mbox{\lr{$0.10$--$0.12$}} برای \lr{AUROC} می‌دهد. آزمون \lr{DeLong} فقط برای مدل‌هایی قابل اتکاست که روی همان نمونۀ آزمون و با بردارهای احتمال هم‌تراز ارزیابی شده باشند. چون همۀ خروجی‌های این پایان‌نامه چنین شرطی را ندارند، تفسیر اصلی بر برآوردهای نقطه‌ای، بازه‌های اطمینان موجود و سلسله‌مراتب پروتکل‌ها تکیه می‌کند؛ آزمون‌های جفتی کامل‌تر مانند \lr{DeLong}، \lr{McNemar} و بوت‌استرپ جفتی برای کار آینده توصیه می‌شوند.

\subsection{استنتاج جایگزین \lr{SOFA} و شاخص شوک}

نمره‌ی \lr{SOFA} (\lr{Sequential Organ Failure Assessment}) اختلال اندام را کمی می‌کند. از جایگزین ساده‌شده استفاده می‌شود: $\text{\lr{SOFA}}_{\text{\lr{proxy}}} = w_1(1 - \text{\lr{creatinine}}) + w_2(1 - \text{\lr{platelets}}) + w_3(1 - \text{\lr{SpO}}_2) + w_4(1 - \text{\lr{lactate}})$ با وزن‌های برابر، هر جزء نرمال‌شده به $[0,1]$. مقادیر بالاتر عملکرد بدتر را نشان می‌دهد. شاخص شوک = ضربان قلب / فشار خون سیستولیک؛ بازه‌ی طبیعی $0.5$-$0.7$؛ مقادیری برای \mbox{\lr{SI}} فراتر از \mbox{\lr{$1.0$}} ناپایداری همودینامیک را نشان می‌دهند. با \lr{clip} و \lr{min-max} به $[0,1]$ نرمال می‌شود. شاخص شوک بالا با ترخیص غیرقابل‌تحقق همراه است.

\section{معماری \lr{V5}: توجه ویژگی دروازه‌ای با نمونه‌های اولیه‌ی متقاطع}
\label{sec:v5_architecture}
%==============================================================================

معماری \lr{V5} دو جزء تازه برای پوشش محدودیت‌های شناسایی‌شده در \lr{V1--V4} معرفی می‌کند:

\subsection{توجه ویژگی دروازه‌ای (\lr{GFA})}

با الهام از \lr{TabNet}~\cite{arik2021tabnet}، ماژول \lr{GFA} انتخاب ویژگی متوالی به‌ازای هر نمونه را پیاده می‌کند. در هر یک از $S=3$ گام توجه، ماژول محاسبه می‌کند:
\begin{align}
\mathbf{a}_s &= \text{\lr{softmax}}(\mathbf{W}_s \cdot \mathbf{h}_{\text{\lr{shared}}} \odot \mathbf{p}_s) \\
\mathbf{p}_{s+1} &= \mathbf{p}_s \cdot (\gamma - \mathbf{a}_s) \\
\mathbf{o}_s &= \text{\lr{FC}}_s(\mathbf{a}_s \odot \mathbf{x})
\end{align}
که در آن $\mathbf{p}_s$ مقیاس پیشین (مقداردهی اولیه به $\mathbf{1}$)، $\gamma=1.5$ ضریب تنک‌سازی، و $\mathbf{h}_{\text{\lr{shared}}} = \text{\lr{FC}}_{\text{\lr{shared}}}(\mathbf{x})$ بافت مشترک است. خروجی تجمیع‌شده $\sum_s \mathbf{o}_s$ با مسیر موازی \lr{transformer} از طریق دروازه‌ی آموخته‌شده ترکیب می‌شود: $\mathbf{z} = \sigma(g) \cdot \mathbf{z}_{\text{\lr{trans}}} + (1-\sigma(g)) \cdot \mathbf{z}_{\text{\lr{GFA}}}$.

طراحی دومسیره به مدل اجازه می‌دهد \emph{محلی} انتخاب ویژگی (\lr{GFA}) را با مدل‌سازی \emph{سراسری} تعامل ویژگی (\lr{transformer}) ترکیب کند و هم اهمیت سطح ویژگی و هم روابط متقاطع ویژگی را بگیرد.

\subsection{سر نمونه‌ی اولیه‌ای متقاطع}

شبکه‌ی نمونه‌ی اولیه‌ای استاندارد شباهت کسینوسی بین بردارهای نهفتۀ پرس‌وجو و میانگین نمونه‌های اولیه‌ی کلاس را محاسبه می‌کند. سر متقاطع \lr{V5} در عوض از خودتوجه متقاطع چندسر استفاده می‌کند:
\begin{equation}
\mathbf{h}_{\text{\lr{cross}}} = \text{\lr{MultiHead}}(\mathbf{Q}=\mathbf{q}, \mathbf{K}=\mathbf{P}, \mathbf{V}=\mathbf{P}) + \mathbf{q}
\end{equation}
که در آن $\mathbf{q}$ بردار نهفتۀ پرس‌وجو (گسترش‌یافته روی کلاس‌ها) و $\mathbf{P}$ نمونه‌های اولیه‌ی کلاس‌اند. سپس \lr{MLP} طبقه‌بند لگاریتم‌های شانس را از $[\mathbf{h}_{\text{\lr{cross}}}; \mathbf{q}]$ تولید می‌کند، با ترکیب $50/50$ با لگاریتم‌های شانس نمونه‌ی اولیه‌ای کسینوسی. این به مدل اجازه می‌دهد توابع شباهت غیرخطی و وابسته به بافت بیاموزد به‌جای تکیه صرف بر معیارهای فاصله.

\section{معماری \lr{V6}: گروه عمیق ناهمگن}
\label{sec:v6_architecture}
%==============================================================================

آزمایش \lr{V6} بررسی می‌کند آیا تنوع معماری عملکرد گروه را بهبود می‌دهد. سه معماری رمزگذار متمایز ترکیب می‌شوند:

\begin{enumerate}
\item \textbf{\lr{FT-Transformer}}: بردارهای نهفته‌سازی آموخته‌شده به‌ازای هر ویژگی ($\mathbf{e}_j = \mathbf{W}_j x_j + \mathbf{b}_j$) که از لایه‌های خودتوجه با تجمیع مبتنی‌بر توجه عبور می‌کنند.
\item \textbf{\lr{GFA-Transformer}}: معماری دومسیره‌ی \lr{V5} که توجه ویژگی دروازه‌ای را با \lr{transformer} ویژگی‌های گروه‌بندی‌شده ترکیب می‌کند.
\item \textbf{\lr{MLP-Mixer}}: معماری بدون توجه با \lr{MLP}های درهم‌تنیده‌ی مخلوط‌کردن توکن (فضایی) و مخلوط‌کردن کانال (ویژگی)، اقتباس‌شده از بینایی به داده‌ی جدولی.
\end{enumerate}

هر معماری با یک سر کم‌نمونه عمومی، شامل نمونه‌ی اولیه‌ای و شبکه‌ی رابطه، ترکیب می‌شود و با همان فرایند چندمرحله‌ای آموزش می‌بیند: \lr{SupCon}، تقطیر دانش نظارت‌شده و آموزش اپیزودیک. فرا یادگیرندۀ نهایی پیش‌بینی‌های \lr{XGBoost} و رگرسیون لجستیک کالیبره‌شده را ترکیب می‌کند. \lr{V6} همچنین از پیش‌بینی مطابق تقسیم‌شده برای سنجش عدم‌قطعیت استفاده می‌کند.

\section{معماری‌های جایگزین مورد بررسی}
%==============================================================================

چند جایگزین پیش از تثبیت طرح فعلی در نظر گرفته شد. \textbf{\lr{MAML} به‌جای نمونه‌ی اولیه‌ای}: \lr{MAML} انعطاف‌پذیرتر اما پرهزینه‌ی محاسباتی است؛ نمونه‌ی اولیه‌ای برای کارایی به‌کار می‌رود. \textbf{\lr{RNN}/\lr{LSTM} به‌جای \lr{transformer}}: \lr{RNN}ها می‌توانند دنباله‌های زمانی را مدل کنند؛ \lr{transformer} برای موازی‌سازی و توجه روی توکن‌ها استفاده می‌شود. \textbf{فقط جدولی خام}: هم بازنمایی جدولی و هم توکن پشتیبانی می‌شود؛ ترکیبی (\lr{ETHOS} + جدولی) از هر کدام بهتر عمل می‌کند. \textbf{$K$ متفاوت}: در تحلیل حساسیت $K \in \{1,3,5,7,10,15,20\}$ متغیر است؛ مدل اصلی $K=5$ دارد. \textbf{\lr{FedProx} به‌جای \lr{FedAvg}}: \lr{FedProx} جمله‌ی مجاورتی برای غیر-\lr{IID} می‌افزاید؛ \lr{FedAvg} به‌عنوان خط پایه و \lr{FedProx} برای کار آینده باقی می‌ماند. این انتخاب‌ها برای بازتولیدپذیری و راهنمایی گسترش‌های آینده مستند شده‌اند.

\section{جزئیات پروتکل ارزیابی}
%==============================================================================

برای مدل‌های کلاسیک، برازش روی آموزش (\pnum{1,280} نمونه)، تنظیم آستانه روی اعتبارسنجی (\pnum{320} نمونه)، و یک‌بار ارزیابی روی آزمون (\pnum{400} نمونه) انجام می‌شود. برای \lr{ETHOS} و یادگیری کم‌نمونه، آموزش اپیزودیک روی آموزش، اعتبارسنجی با $100$ اپیزود روی اعتبارسنجی، و ارزیابی روی آزمون داریم. ارزیابی آزمون برای \lr{ETHOS} در این پایان‌نامه دو صورت دارد: (\pnum{1}) ارزیابی ثابت روی کل مجموعۀ آزمون در \lr{Exp1} که برای \texttt{\lr{Proposed\_FewShot}} عدد منجمد \pnum{0.611} را گزارش می‌کند؛ (\pnum{2}) نمونه‌برداری اپیزودیک ($100$ اپیزود، هر کدام با $K$ پشتیبان و $Q$ پرس‌وجو از آزمون) که برای حساسیت \lr{K}-شات به‌کار می‌رود و \lr{AUROC} را در بازۀ \prange{0.557}{0.564} نشان می‌دهد. دو پروتکل مقادیر مطلق متفاوتی می‌دهند؛ بنابراین در هر آزمایش صریحاً مشخص می‌کنیم کدام پروتکل مبنا بوده است. برای ارزیابی فدرال، مدل سراسریِ تجمیع‌شده روی گره‌ها بر کل مجموعۀ آزمون اعمال می‌شود.
