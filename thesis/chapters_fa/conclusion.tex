\chapter{نتیجه‌گیری}
\label{ch:conclusion}

این پایان‌نامه مسئلۀ پیش‌بینی امکان‌پذیری درمان را بر پایۀ دادۀ بالینی محدود در \lr{MIMIC\text{-}IV} بررسی کرد. در این پژوهش، امکان‌پذیری درمان به‌صورت عملیاتی به‌معنای پیش‌بینی پیامد قابل‌اندازه‌گیری درمان/ترخیص از سیگنال‌های اولیه \lr{ICU} تعریف شد و نه به‌معنای توصیه مستقیم درمان، نسخه‌نویسی خودکار یا جایگزینی تصمیم پزشک. چارچوب پیشنهادی از توکن‌های علائم، رمزگذار \lr{ETHOS}، یادگیری کم‌نمونه، آموزش فدرال و چند روش تکمیلی مانند \lr{Proto-MAML}، تقطیر دانش، یادگیری تضادی نظارت‌شده، توجه درگاهی به ویژگی، \lr{Cross-Attention} و انباشت استفاده می‌کند. اعداد اصلی این فصل بر خروجی‌های منجمد مخزن استوارند و مسیر توسعه‌ی \lr{V1--V6} در فصل~\ref{ch:results} به‌عنوان سابقۀ تکامل روش گزارش شده است. یافته‌های اصلی عبارت‌اند از:

\begin{itemize}
\item \textbf{سقف جدولی (منجمد):} فرایند اختصاصی جدولی به \lr{AUROC} آزمون \textbf{\pnum{0.748}} می‌رسد (فصل~\ref{ch:results}، جدول~\ref{tab:baselines}).
\item \textbf{انباشت کم‌نمونه + کلاسیک (منجمد):} بهترین اجرای انباشتی به \lr{AUROC} \textbf{\pnum{0.746}} \cnum{[0.699, 0.797]} می‌رسد؛ یعنی تنها حدود \textbf{\pnum{0.002}} پایین‌تر از برآورد نقطه‌ای سقف.
\item \textbf{تنها کم‌نمونه (منجمد):} بهترین کم‌نمونه (\lr{GFA}+\lr{CrossAttn}) به \lr{AUROC} \textbf{\pnum{0.670}} \cnum{[0.612, 0.726]} می‌رسد.
\item \textbf{مقایسۀ \lr{Exp1} (تقسیم ثابت):} مدل پیشنهادی کم‌نمونه با \textbf{\pnum{0.611}} در برابر \lr{Random Forest} با \textbf{\pnum{0.762}} روی همان چارچوب ارزیابی.
\item \textbf{کمپین \lr{V} (تاریخی):} جدول~\ref{tab:improved_strategies} بهبودهای تدریجی را گزارش می‌کند (مثلاً انباشت \lr{V4} در \pnum{0.732}، بهترین بذر \lr{V5} در \pnum{0.737}، و \lr{RF} با آستانۀ تنظیم‌شده در \pnum{0.760})؛ برای ادعاهای اجرایی نخست به جداول اصلی ارجاع دهید.
\item \textbf{یادگیری فدرال:} شکاف \lr{AUROC} میان آموزش متمرکز و فدرال در آزمایش گزارش‌شده \textbf{\pnum{0.005}} است؛ یعنی آموزش حافظ حریم خصوصی در این شبیه‌سازی با هزینۀ عملکردی محدود امکان‌پذیر بوده است.
\item \textbf{پیش‌بینی مطابق (\lr{conformal}) (اکتشافی):} تحلیل‌های سبک \lr{V6} مجموعه‌های پیش‌بینی کالیبره در اجراهای توسعه فراهم می‌کنند؛ اینجا به‌عنوان شاخص‌های کلیدی ارسال دوباره تأکید نمی‌شوند.
\item \textbf{سقف تجربی:} مدل‌های جدولی قوی در این کوهورت، بسته به پروتکل، در بازۀ \textbf{\prange{0.748}{0.760}} قرار دارند.
\end{itemize}

این پژوهش یک مطالعۀ بازتولیدپذیر از یادگیری کم‌نمونه جدولی برای امکان‌پذیری درمان روی \lr{MIMIC\text{-}IV} ارائه می‌کند؛ همراه با خطوط پایۀ صادقانه، بازنمایی بالینی مبتنی بر توکن‌های علائم، ارزیابی فدرال و خروجی‌های قابل بازتولید در \texttt{\lr{results/}} و جداول تولیدشده‌ی \lr{LaTeX}. مقدار \pnum{0.762} برای \lr{Random Forest} به پروتکل \lr{Exp1} و مقدار \pnum{0.748} به فرایند منجمد \lr{Tabular SOTA} مربوط است؛ ازاین‌رو تفاوت آن‌ها ناشی از تفاوت پروتکل است. عملکرد انباشتی نزدیک به سقف، یعنی \textbf{\pnum{0.746}} در برابر \textbf{\pnum{0.748}}، نشان می‌دهد اگر بازنمایی، هم‌جوشی مدل‌ها و روند آموزش با ماهیت وظیفه هماهنگ شود، روش‌های کم‌نمونه می‌توانند به مدل‌های کلاسیک قوی نزدیک شوند؛ اما مدل کم‌نمونه خام جایگزین مستقیم \lr{Random Forest} نیست.

\subsection{پیامدها برای استقرار بالینی}

برای بالین‌دانان و مدیران سلامت، یافته‌ها در مرحلۀ \textbf{پژوهشی} هستند:

\begin{enumerate}
\item \textbf{عملکرد:} مدل انباشتی منجمد به \lr{AUROC} \textbf{\pnum{0.746}} می‌رسد؛ پیش از هر کاربرد پشتیبان تصمیم، اعتبارسنجی محلی الزامی است.
\item \textbf{آموزش حافظ حریم خصوصی:} آموزش فدرال در شبیه‌سازی ما هزینۀ کوچکی در \lr{AUROC} دارد؛ با این حال، استقرار چندسایتی واقعی به آزمون عملیاتی نیاز دارد.
\item \textbf{سازگاری:} سازگاری به‌سبک \lr{Proto-MAML} امیدبخش است اما باید روی دادۀ آینده‌نگر و ویژۀ سایت ارزیابی شود.
\item \textbf{شفافیت:} انباشت امکان مقایسۀ اجزای کم‌نمونه در برابر کلاسیک را وقتی هر دو در دسترس‌اند فراهم می‌کند.
\item \textbf{توصیه‌های استقرار:} از آستانه‌های اطمینان، بررسی‌های انصاف، پایش انحراف، و نظارت انسانی استفاده کنید؛ قضاوت بالینی را جایگزین نکنید.
\end{enumerate}

\subsection{تطبیق نهایی با پروپوزال}

در مقایسه با پروپوزال اولیه، هستۀ علمی کار حفظ شده است: پیش‌بینی امکان‌پذیری درمان با ترکیب یادگیری کم‌نمونه، یادگیری فدرال، \lr{ETHOS} و توکن‌های علائم. تفاوت اصلی در دامنه‌ی عملیاتی است. پروپوزال از داده‌های چندمنبعی گسترده و حتی سناریوهای دارای واسط کاربری سخن می‌گفت؛ اجرای نهایی، برای حفظ بازتولیدپذیری و اعتبار علمی، روی \lr{MIMIC\text{-}IV} و منابع درون \lr{EHR} یعنی آزمایشگاه‌ها، علائم حیاتی و ویژگی‌های مشتق‌شده متمرکز شد. بنابراین این پایان‌نامه نه یک سامانۀ تولیدی کامل، بلکه یک ارزیابی پژوهشی دقیق و قابل بازتولید از چارچوب پیشنهادی است. پیاده‌سازی واسط کاربری، اتصال به داده‌ی زنده یا پوشیدنی، و اعتبارسنجی چندسایتی واقعی به‌عنوان گام‌های طبیعی پس از این پایان‌نامه باقی می‌مانند.

\subsection{توصیه‌ها برای پژوهشگران و عملگران}

\begin{enumerate}
\item \textbf{برای بیشینۀ دقت در این کوهورت:} فرایند جدولی \lr{SOTA} منجمد (\lr{AUROC} \textbf{\pnum{0.748}}) یا خطوط پایۀ کلاسیک هم‌تراز را ترجیح دهید؛ انباشت‌های کم‌نمونه را نیز گزارش کنید.
\item \textbf{برای هم‌جوشی کم‌نمونه + کلاسیک:} بهترین پیکربندی انباشتی مستندشده را به‌عنوان مرجع (\textbf{\pnum{0.746}}) به کار ببرید.
\item \textbf{برای تنظیمات فدرال:} در چیدمان ما اختلاف‌های عملکرد کوچک در برابر متمرکز انتظار دارید؛ ارتباط و ناهمگونی داده را جداگانه اعتبارسنجی کنید.
\item \textbf{برای سایت‌های جدید:} کالیبراسیون، تنظیم آستانه، و ارزیابی آینده‌نگر را برنامه‌ریزی کنید.
\item \textbf{برای ادعاهای غیربرتری آماری:} آزمون‌های آماری جفتی روی بردارهای پیش‌بینی هم‌تراز اجرا کنید؛ به $p$-مقادیر تاریخی از پیش‌نویس‌های پیشین تکیه نکنید.
\end{enumerate}

\subsection{اهمیت برای حوزه}

این کار به انفورماتیک بالینی و فرا یادگیری سه افزوده‌ی اصلی دارد: (\pnum{1}) گزارش هم‌زمان سقف‌های جدولی قوی و مسیر کم‌نمونه نزدیک به آن‌ها (\textbf{\pnum{0.746}} در برابر \textbf{\pnum{0.748}})؛ (\pnum{2}) ترکیب دیدگاه فدرال و کم‌نمونه در یک وظیفۀ ساخت‌یافتۀ \lr{ICU}؛ و (\pnum{3}) مستندسازی مسیر بهبود چندمرحله‌ای \lr{V1--V6} با تفکیک روشن میان اعداد منجمد و اعداد تاریخی. توکن‌های علائم و ترکیب \lr{ETHOS} با یادگیری کم‌نمونه می‌توانند قالبی برای سناریوهای کم‌داده و حساس به حریم خصوصی باشند، به‌شرط آنکه در داده‌های بیرونی نیز اعتبارسنجی شوند.

\subsection{جمع‌بندی}

شکاف میان پیکربندی‌های ساده‌ی کم‌نمونه و مدل‌های جدولی قوی واقعی است؛ برای نمونه در \lr{Exp1}، مقدار \textbf{\pnum{0.611}} در برابر \textbf{\pnum{0.762}} قرار می‌گیرد. با این حال، مهندسی سیستماتیک در بازنمایی، مراحل آموزش، تقطیر، سازگاری و انباشت می‌تواند اجرای یکپارچه را به برآورد نقطه‌ای \lr{SOTA} جدولی نزدیک کند. بهبودهای بعدی به‌احتمال زیاد به پیش‌آموزش بزرگ‌تر، دادۀ چندسایتی و اعتبارسنجی آینده‌نگر نیاز خواهند داشت.
