\chapter{نتایج}
\label{ch:results}

این فصل نتایج تجربی را به‌ترتیب اهمیت گزارش می‌کند. اعداد و معیارهای اصلی همان خروجی‌های ثبت‌شدۀ آزمایش‌ها هستند و در این بخش فقط نحوۀ خواندن و تفسیر آن‌ها روشن‌تر می‌شود.

\paragraph{سلسله‌مراتب گزارش‌دهی (چگونه اعداد را بخوانیم).}
در این پایان‌نامه چند معیار \textbf{مکمل} گزارش می‌شود. هر کدام به پرسش متفاوتی پاسخ می‌دهد و بنابراین نباید به‌جای دیگری تفسیر شود:
\begin{enumerate}[leftmargin=2em]
  \item \textbf{سقف جدولی}: \lr{Random Forest} و \lr{XGBoost} تنظیم‌شده و فرایند \lr{SOTA} جدولی، همراه با افزایش داده و تنظیم آستانه. این بخش مرز بالای تجربی مدل‌های کلاسیک را روی همین کوهورت نشان می‌دهد.
  \item \textbf{پروتکل \lr{Exp1}}: در آزمایش اصلی \lr{Exp1}، مدل پیشنهادی کم‌نمونه و خطوط پایه در یک روند مشترک مقایسه می‌شوند. این اعداد به همان پیکربندی آموزش اپیزودیک مخزن مربوط‌اند و جداول خروجی نیز به‌صورت خودکار در پیوست~\ref{app:pipeline_tables} آمده‌اند.
  \item \textbf{ترکیب جدولی و بهترین اجرای کم‌نمونه}: بردارهای نهفتۀ \lr{ETHOS} همراه با ویژگی‌های جدولی و بهترین فرایند یکپارچه‌ی کم‌نمونه/انباشتی گزارش می‌شوند. این اجراها معمولاً بخش مهمی از فاصله تا سقف جدولی را، نسبت به اعداد اپیزودیک ساده، کاهش می‌دهند.
  \item \textbf{فدرال در برابر متمرکز}: این بخش بده‌بستان میان حریم خصوصی و سودمندی را در اجرای فدرال جداگانه گزارش می‌کند.
\end{enumerate}
\textbf{حساسیت \lr{K}-شات} در \lr{Exp2} با پروتکل \emph{اپیزودیک} ارزیابی می‌شود. بنابراین \lr{AUROC} آن را نباید مستقیماً با \lr{AUROC} کل مجموعۀ آزمون در \lr{Exp1} یا با نتایج انباشتی مقایسه کرد. این بخش فقط نشان می‌دهد اندازۀ مجموعۀ پشتیبان چه اثری بر عملکرد اپیزودیک دارد.

\section{\texorpdfstring{عملکرد \lr{ETHOS} و یادگیری کم‌نمونه}{عملکرد ETHOS و یادگیری کم‌نمونه}}
%==============================================================================

جدول~\ref{tab:ethos_metrics} معیارهای بهترین اجرای یکپارچۀ کم‌نمونه/انباشتی را نشان می‌دهد. این اجرا شامل یک \lr{Ensemble} در مسیر کم‌نمونه (ترکیب \lr{GFA}، \lr{Cross-Attention} و سازگاری به‌سبک \lr{Proto-MAML}) و یک \textbf{فرا یادگیر} انباشتی است که احتمال‌های کم‌نمونه، احتمال‌های \lr{RF} و ویژگی‌های نهفته‌سازی را ترکیب می‌کند. اعداد جدول، برآوردهای نقطه‌ای مجموعۀ آزمون برای همان اجرای ثبت‌شده‌اند.

\begin{table}[htbp]
\centering

\textbf{فاصله تا سقف جدولی \lr{SOTA}} در جدول~\ref{tab:baselines} بسیار کوچک است: اختلاف \lr{AUROC} بین \lr{Stacked BEST} و سقف جدولی برابر \textbf{\pnum{0.002}} است (\cnum{0.748 - 0.746 = 0.002}). این مقدار در حد نوسان اجراهاست. با این حال، زیرمدل کم‌نمونه به‌تنهایی با \pnum{0.670} هنوز پایین‌تر از سقف قرار می‌گیرد؛ همین شکاف دلیل اصلی استفاده از انباشت و طراحی ترکیبی است. همچنین باید توجه کرد که مقدار \pnum{0.762} برای \lr{Random Forest} به پروتکل ثابت \lr{Exp1} تعلق دارد، در حالی که مقدار \pnum{0.748} به فرایند منجمد \lr{Tabular SOTA} مربوط است؛ این دو عدد پروتکل‌های متفاوت را خلاصه می‌کنند و تناقض محسوب نمی‌شوند. شکل~\ref{fig:learning_curves} نشان می‌دهد \lr{RF} و \lr{ETHOS} با افزایش اندازۀ آموزش در مجموعۀ تشخیصی چگونه مقیاس می‌یابند.

\section{مطالعات موردی بالینی}
%==============================================================================

برای نمایش رفتار مدل، سه مورد نماینده از مجموعۀ آزمون ارائه می‌کنیم:

\textbf{مورد \pnum{1} (مثبت واقعی):} مرد \pnum{62} ساله با سپسیس. کراتینین \pnum{1.2} \lr{mg/dL} (نرمال‌شده \pnum{0.4})، روند بهبود؛ لاکتات \pnum{1.8} (\pnum{0.35})؛ علائم حیاتی پایدار. \lr{RF} امکان‌پذیری را \pnum{0.72} پیش‌بینی کرد؛ ترخیص واقعی به بازتوانی. در تحلیل \lr{SHAP}، کراتینین با اثر \cnum{-0.08}، لاکتات با اثر \cnum{-0.05}، و \lr{SOFA} با اثر \cnum{-0.04} به تصمیم درست مدل کمک کردند.

\textbf{مورد \pnum{2} (منفی کاذب):} زن \pnum{58} ساله با تشخیص قلبی. کراتینین مرزی (\pnum{0.45} نرمال‌شده)؛ \lr{RF} غیرممکن (\pnum{0.48}) پیش‌بینی کرد اما بیمار به منزل ترخیص شد. خطا احتمالاً به‌دلیل کراتینین در بازۀ مبهم؛ زمینۀ اضافی (مثلاً روند بهبود) می‌توانست کمک کند.

\textbf{مورد \pnum{3} (مثبت کاذب):} مرد \pnum{71} ساله با نارسایی کلیه. کراتینین بالا (\pnum{0.72} نرمال‌شده)؛ \lr{RF} امکان‌پذیری را \pnum{0.58} پیش‌بینی کرد اما بیمار نیاز به اقامت طولانی داشت. مدل ممکن است تعامل سن-کراتینین را کم‌وزن کرده باشد.

این موارد برجسته می‌کنند که مقادیر آزمایشگاهی مرزی و زیرگروه‌های نادر همچنان چالش‌زا هستند. در استقرار باید پیش‌بینی‌های با اطمینان پایین برای بازبینی بالینی علامت‌گذاری شوند.

\section{کالیبراسیون و تحلیل حساسیت}

\section{تحلیل خطا به‌ازای ویژگی}
%==============================================================================

تحلیل کردیم کدام ویژگی‌ها با طبقه‌بندی اشتباه بیشتر همراه‌اند. برای مثبت‌های کاذب (امکان‌پذیری پیش‌بینی‌شده، غیرممکن واقعی): کراتینین و لاکتات اغلب در بازۀ پایین-نرمال‌اند و مدل را به کم‌برآوردن ریسک سوق می‌دهند. برای منفی‌های کاذب (غیرممکن پیش‌بینی‌شده، امکان‌پذیر واقعی): سن و نمرۀ \lr{SOFA} گاهی بالا هستند با وجود ترخیص مطلوب؛ مدل ممکن است بیش‌محافظه‌کار باشد. این الگو نشان می‌دهد افزودن اطلاعات روند (بهبود در برابر وخامت) و یادداشت‌های بالینی می‌تواند خطاها را کاهش دهد. بازنمایی فعلی از تجمیع \pnum{24} ساعت اول استفاده می‌کند و پویایی زمانی را نمی‌گیرد.

%==============================================================================
\section{تحلیل \lr{ROC} و دقت-بازخوانی}
%==============================================================================

منحنی \lr{ROC} نرخ مثبت واقعی (حساسیت) را در برابر نرخ مثبت کاذب ($1-$ اختصاصیت) در آستانه‌های مختلف نشان می‌دهد. \lr{AUROC} برابر \pnum{0.76} یعنی در حدود \pnum{76}\% موارد، مدل به یک بیمار واقعاً امکان‌پذیر امتیاز بالاتری از یک بیمار واقعاً ناممکن می‌دهد. در دادۀ نامتعادل، منحنی دقت-بازخوانی نیز اهمیت دارد. در آستانۀ پیش‌فرض \pnum{0.54}، مدل \lr{RF} دقت \pnum{0.67} و بازخوانی \pnum{0.69} دارد. برای استقرار، آستانه باید بر اساس هزینۀ مثبت کاذب، مانند ترخیص زودهنگام، و منفی کاذب، مانند تأخیر در ترخیص، تنظیم شود. آستانۀ پایین‌تر بازخوانی را افزایش می‌دهد اما هشدارهای کاذب بیشتری تولید می‌کند؛ بنابراین نقطۀ قابل قبول باید با مشارکت ذی‌نفعان بالینی و بر اساس اولویت‌های هر مؤسسه تعیین شود.

\section{منحنی‌های یادگیری و کارایی داده}
%==============================================================================

مجموعۀ آموزش به \pnum{25}\%، \pnum{50}\%، \pnum{75}\% و \pnum{100}\% زیرنمونه‌برداری شد و \lr{AUROC} آزمون اندازه‌گیری گردید. برای \lr{Random Forest} مقادیر به‌ترتیب \pnum{0.68}، \pnum{0.72}، \pnum{0.75} و \pnum{0.76} بود. برای \lr{ETHOS} همراه با یادگیری کم‌نمونه نیز مقادیر \pnum{0.55}، \pnum{0.58}، \pnum{0.60} و \pnum{0.62} به‌دست آمد. هر دو مدل از افزایش داده سود می‌برند، اما شیب بهبود \lr{RF} بیشتر است؛ یعنی در این وظیفه، مدل کلاسیک از داده‌ی موجود کاراتر استفاده می‌کند. با تنها \pnum{25}\% داده‌ی آموزش، یعنی \pnum{320} نمونه، \lr{RF} هنوز به \lr{AUROC} برابر \pnum{0.68} می‌رسد که برای تنظیمات کم‌داده قابل قبول است. در مقابل، \lr{ETHOS} برای رسیدن به عملکرد مشابه به داده‌ی بیشتری نیاز دارد. این نتیجه استفاده از مدل‌های کلاسیک را در سناریوهایی که داده‌ی کافی وجود دارد تقویت می‌کند و نقش کم‌نمونه را بیشتر برای سناریوهای فدرال یا واقعاً کم‌داده برجسته می‌سازد.

\section{راهبردهای آموزش بهبودیافته}
\label{sec:improved-training}

\begin{quote}
\textbf{یادداشت خواننده (اسنپ‌شات ارسال).} زیربخش‌های زیر مسیر توسعۀ گستردۀ \lr{V1--V6} را \emph{توسعه} توصیف می‌کنند (روش‌شناسی و بنچمارک‌های میانی آن کمپین). \textbf{اعدادی که برای این پایان‌نامه منجمد شده‌اند} جداول همگام‌شده با مخزن در ابتدای فصل~\ref{ch:results} است (\lr{AUROC} آزمون جدولی \lr{SOTA} \textbf{\pnum{0.748}}؛ اجرای انباشتی یکپارچه \textbf{\pnum{0.746}} \cnum{[0.699, 0.797]}؛ مدل پیشنهادی کم‌نمونه در \lr{Exp1} با \textbf{\pnum{0.611}}؛ و غیره) و پیوست~\ref{app:pipeline_tables}. جایی که روایت به \pnum{0.732}، \pnum{0.737}، \pnum{0.760} استناد می‌کند، مگر همان مقدار در آن جداول اولیه ظاهر شود، آن را خروجی \emph{تاریخی} کمپین \lr{V} بدانید.
\end{quote}

بر اساس تحلیل محدودیت‌های مدل نمونه‌اولیه، سه راهبرد بهبود مکمل توسعه دادیم. جدول~\ref{tab:improved_strategies} نتایج را خلاصه می‌کند.

\subsection{راهبرد \pnum{1}: سر نمونه‌اولیه-رابطۀ ترکیبی با پیش‌تنظیم نظارت‌شده}

شبکۀ نمونه‌اولیه شباهت را از راه فاصلۀ کسینوسی تا مرکزهای کلاس محاسبه می‌کند که ضمناً مرزهای محدب کلاس در فضای نهفته فرض می‌کند. برای شل کردن این فرض، \textbf{سر شبکۀ رابطه} (\lr{MLP} کوچک که تابع شباهت غیرخطی بین بردارهای نهفتۀ پرس‌وجو و نمونه‌های اولیۀ کلاس یاد می‌گیرد) معرفی شد. معماری نهایی از \textit{ترکیب درگاهی} هر دو لگاریتم فاصلۀ کسینوسی نمونه‌اولیه و لگاریتم‌های رابطۀ آموزش‌دیده استفاده می‌کند:
\begin{equation}
\text{\lr{logits}} = \sigma(g) \cdot \text{\lr{logits}}_{\text{\lr{proto}}} + (1 - \sigma(g)) \cdot \text{\lr{logits}}_{\text{\lr{relation}}}
\end{equation}
که در آن $g$ پارامتر درگاه قابل‌یادگیری و $\sigma$ تابع سیگموئید است.

همچنین \textbf{پیش‌تنظیم نظارت‌شده رمزگذار} افزودیم: پیش از آموزش اپیزودیک، رمزگذار \lr{ETHOS} ابتدا با زیان آنتروپی متقاطع نظارت‌شده روی کل مجموعۀ آموزش (با هموارسازی برچسب \pnum{0.05}) آموزش می‌بیند. این به رمزگذار بازنمایی‌های اولیۀ بهتری برای فاز اپیزودیک بعدی می‌دهد.

جزئیات آموزش در دو مرحله اجرا شد: ابتدا \pnum{40} دوره پیش‌تنظیم نظارت‌شده با \lr{AdamW}، نرخ یادگیری \LTRformula{5{\times}10^{-4} } و وزن‌زدایی \LTRformula{10^{-4} }. سپس \pnum{1,500} اپیزود آموزش اپیزودیک با افزایش داده اجرا شد (نویز گاوسی $\sigma=0.03$، ماسک \pnum{5}\% ویژگی). ارزیابی از \lr{Ensemble} \pnum{80} اپیزود با مجموعه‌های پشتیبان متفاوت به‌همراه یک ارزیابی پشتیبان کامل استفاده می‌کند.

\subsection{راهبرد \pnum{2}: ترکیبی بهبودیافته (\lr{ETHOS} + \lr{XGBoost})}

\lr{Random Forest} در معماری ترکیبی را با \lr{XGBoost} جایگزین کردیم، با \lr{RandomizedSearchCV} \pnum{50} آزمون و اعتبارسنجی متقاطع \pnum{5}‌لایۀ طبقه‌ای. ترکیب، بردارهای نهفتۀ \pnum{128}بعدی \lr{ETHOS} (از رمزگذار پیش‌تنظیم‌شده نظارت‌شده) را با ویژگی‌های جدولی اصلی \pnum{25}بعدی می‌چسباند و ورودی \pnum{153}بعدی می‌سازد. تقویت گرادیان \lr{XGBoost} اطلاعات مکمل بین بازنمایی‌های آموزش‌دیده و ویژگی‌های خام را بهتر بهره می‌برد. تنظیم آستانه روی اعتبارسنجی \lr{F1} را بیشینه می‌کند.

\subsection{راهبرد \pnum{3}: طبقه‌بند نظارت‌شده \lr{ETHOS}}

به‌عنوان مرز بالای توان رمزگذار \lr{ETHOS}، سر طبقه‌بند \pnum{3}لایه مستقیماً روی بردارهای نهفتۀ رمزگذار ($128{\to}128{\to}64{\to}2$) با وزن‌دهی کلاس، هموارسازی برچسب، و توقف زودهنگام بر اساس \lr{AUROC} اعتبارسنجی آموزش دادیم. این پارادایم آموزش اپیزودیک را به‌طور کامل دور می‌زند.

\subsection{راهبرد \pnum{4}: مدل کم‌نمونه \lr{V2} با \lr{SupCon} و آموزش چندفازی}

با اتکا به بینش‌های راهبردهای \prange{1}{3}، نسخۀ جامع \lr{V2} با پنج نوآوری کلیدی ساخته شد:

\begin{enumerate}
\item \textbf{مهندسی ویژگی}: گسترش ویژگی‌های اصلی \pnum{22}بعدی با نسبت‌های جفتی، حاصل‌ضرب‌های تعاملی، و جملات مربع، فضای ویژگی را به \pnum{55} بعد می‌رساند. این به رمزگذار سیگنال غنی‌تری برای یادگیری می‌دهد.
\item \textbf{پیش‌آموزش \lr{SupCon}}: به‌جای آنتروپی متقاطع استاندارد، ابتدا رمزگذار را با زیان مقابله‌ای که صریحاً بردارهای نهفتۀ هم‌کلاس را نزدیک و غیرهم‌کلاس را دور می‌کند آموزش می‌دهیم. این فضای نهفته خوشه‌شده می‌سازد؛ دقیقاً هندسی که شبکه‌های نمونه‌اولیه نیاز دارند.
\item \textbf{آموزش چندفازی}: آموزش در سه فاز انجام می‌شود: (الف)~پیش‌آموزش \lr{SupCon} (\pnum{25} دوره) $\to$ (ب)~تنظیم دقیق نظارت‌شده با زیان کانونی و وزن‌دهی کلاس (\pnum{20} دوره با توقف زودهنگام مبتنی بر اعتبارسنجی) $\to$ (ج)~فرا آموزش اپیزودیک (\pnum{1,500} اپیزود) با افزایش نویز گاوسی و برنامه‌ریزی درسی $K$.
\item \textbf{پالایش نمونه‌اولیه ترابردی}: در استنتاج، پالایش \lr{soft k-means} (\pnum{10} تکرار) با آمار مجموعۀ پرس‌وجوی بدون برچسب برای تنظیم نمونه‌های اولیه اعمال می‌کنیم و عملاً به توزیع آزمون سازگار می‌شویم.
\item \textbf{تجمیع چندبذری}: دو مدل با بذرهای متفاوت آموزش‌دیده با پروتکل‌های ارزیابی پشتیبان کامل و اپیزود نمونه‌برداری‌شده ترکیب می‌شوند تا پیش‌بینی تاب‌آور شود.
\end{enumerate}

معماری رمزگذار از تجمیع مبتنی بر توجه (به‌جای توکن \lr{CLS}) با بردارهای نهفتۀ موقعیتی استفاده می‌کند و سر کم‌نمونه لگاریتم‌های نمونه‌اولیه کسینوسی و شبکۀ رابطه را از راه درگاه آموزش‌دیده ترکیب می‌کند.

\subsection{راهبرد \pnum{5}: مدل کم‌نمونه \lr{V3} با تقطیر دانش از جنگل تصادفی}

ایده‌ی اصلی در \lr{V3} این است که معلم \lr{Random Forest} با \lr{AUROC} برابر \pnum{0.760} دانشی دارد که می‌تواند از راه تقطیر دانش (\lr{KD}) به رمزگذار کم‌نمونه منتقل شود. به‌جای یادگیری صرفاً از برچسب‌های سخت، رمزگذار خروجی‌های نرم \lr{RF} را نیز می‌آموزد و مرزهای تصمیم \lr{RF} را به فضای نهفته فشرده منتقل می‌کند. \lr{V3} نسبت به \lr{V2} سه نوآوری افزود:

\begin{enumerate}
\item \textbf{تقطیر دانش}: در هر دو فاز \lr{SupCon} و نظارت‌شده، رمزگذار با زیان ترکیبی آموزش می‌بیند: $\mathcal{L} = \mathcal{L}_{\text{\lr{task}}} + \alpha \cdot \mathcal{L}_{\text{\lr{KD}}}$، که $\mathcal{L}_{\text{\lr{KD}}} = T^2 \cdot \text{\lr{KL}}(\sigma(\mathbf{z}_s/T) \| \sigma(\mathbf{z}_t/T))$ واگرایی \lr{KL} بین پیش‌بینی‌های نرم شاگرد و معلم در دمای $T$ است. از $T=3{-}4$ و $\alpha=0.4{-}0.5$ استفاده می‌کنیم.
\item \textbf{مهندسی ویژگی غنی‌تر}: فضای ویژگی از \pnum{22} به \pnum{92} بعد با نسبت‌های جفتی، حاصل‌ضرب‌ها، جملات مربع، تبدیل‌های لگاریتمی، و آمار خلاصه به‌ازای نمونه (میانگین، انحراف معیار، میانه، شمار مقادیر بالا/پایین) گسترش یافت.
\item \textbf{افزایش داده در زمان آزمون (\lr{TTA})}: در ارزیابی، \pnum{5} نسخۀ افزایش‌یافته از هر نمونۀ آزمون (نویز گاوسی $\sigma=0.015$) ساخته و پیش‌بینی‌ها روی ورودی‌های افزایش‌یافته و اصلی میانگین می‌شوند تا واریانس کاهش یابد.
\end{enumerate}

تجمیع سه‌بذری (بذرهای \pnum{42}، \pnum{49}، \pnum{55}) \lr{AUROC} پایدار به‌ازای بذر \prange{0.701}{0.706} دارد و \lr{Ensemble} به \pnum{0.712} می‌رسد. عملکرد بهترین مدل تکی به‌ازای بذر از قبل با \lr{Ensemble} \lr{V2} (\pnum{0.689}) رقابتی است و نشان می‌دهد تقطیر دانش به‌تنهایی حدود \pnum{0.015} بهبود \lr{AUROC} می‌دهد.

\subsection{راهبرد \pnum{6}: مدل کم‌نمونه \lr{V4} با \lr{Proto-MAML} و فرا یادگیر انباشتی}
\label{sec:v4-strategy}

\lr{V4} سه تکنیک نوآورانه برای \textbf{امکان استقرار بالینی} معرفی می‌کند:

\begin{enumerate}
\item \textbf{\lr{Proto-MAML} (سازگاری زمان آزمون)}: در زمان ارزیابی، مدل آموزش‌دیده را کلون می‌کنیم و \pnum{8} گام گرادیان روی مجموعۀ پشتیبان پیش از پیش‌بینی روی پرس‌وجو انجام می‌دهیم. این پارامترهای مدل (به‌ویژه دما، بایاس کلاس، و شبکۀ رابطه) را به هر وظیفۀ ارزیابی خاص سازگار می‌کند. \lr{Proto-MAML} رویکرد استاندارد در یادگیری کم‌نمونه است و امکان تنظیم دقیق به هر کوهورت بیمار را فراهم می‌کند. حلقۀ درونی از \lr{SGD} با نرخ یادگیری \pnum{0.01} برای پارامترهای سر و $10^{-4}$ برای رمزگذار با برش گرادیان \pnum{1.0} استفاده می‌کند.

\item \textbf{ترنسفورمر ویژگی گروه‌بندی‌شده}: به‌جای قالب توکن‌های علائم \lr{ETHOS} اصلی (که توکن‌های \pnum{4}بعدی خاص می‌طلبد)، از یک رمزگذار ترنسفورمر تمیز استفاده می‌کنیم که \pnum{92} ویژگی مهندسی‌شده را در توکن‌های اندازۀ \pnum{4} گروه‌بندی می‌کند، بردارهای نهفتۀ موقعیتی آموزش‌دیده اعمال می‌کند، و از \pnum{2} لایۀ خودتوجه با تجمیع مبتنی بر توجه عبور می‌دهد. این معماری به قالب ورودی بی‌طرف است و از مکانیزم‌های ضرب شدت و بایاس هم‌وقوعی که رمزگذار \lr{ETHOS} اصلی را به ساختار توکن خاص محدود می‌کند، اجتناب می‌کند.

\item \textbf{فرا یادگیر انباشتی}: یک رگرسیون لجستیک کالیبره‌شده (با کالیبراسیون ایزوتونیک، اعتبارسنجی متقاطع \pnum{3}‌لایه) پنج سیگنال ورودی را ترکیب می‌کند: (الف)~احتمالات تجمیع کم‌نمونه، (ب)~احتمالات \lr{RF}، (ج)~حاصل‌ضرب آن‌ها (سیگنال توافق)، (د)~قدرمطلق تفاوت (سیگنال اختلاف)، و (ه)~\pnum{10} مؤلفۀ اول \lr{PCA} از بردارهای نهفتۀ میانگین رمزگذار. این فرا یادگیر قوت‌های مکمل کم‌نمونه و کلاسیک را می‌آمیزد: مدل کم‌نمونه هندسۀ فضای نهفته را می‌گیرد و به ساختار وظیفه سازگار می‌شود، در حالی که \lr{RF} در افراز مستقیم ویژگی عالی است.
\end{enumerate}

پروتکل ارزیابی استنتاج پشتیبان کامل استاندارد (وزن \LTRformula{2\times})، سازگاری پشتیبان کامل \lr{Proto-MAML} (وزن \LTRformula{3\times})، پالایش نمونه‌اولیه ترابردی (وزن \LTRformula{2\times})، \pnum{3} عبور \lr{TTA}، و \pnum{20} اپیزود نمونه‌برداری‌شده (\pnum{3} تا با سازگاری \lr{MAML}) را ترکیب می‌کند. \lr{AUROC} به‌ازای بذر از \pnum{0.696} تا \pnum{0.713} متغیر است. مدل انباشتی به \lr{AUROC} \textbf{\pnum{0.732}} \cnum{[0.684, 0.783]} می‌رسد؛ یعنی از نظر توصیفی به خط پایۀ \lr{RF} تاریخی همان کمپین (\pnum{0.735}) نزدیک می‌شود، هرچند برای ادعای تساوی آماری به آزمون جفتی روی بردارهای پیش‌بینی هم‌تراز نیاز است.

\subsection{راهبرد \pnum{7}: مدل کم‌نمونه \lr{V5} با توجه دروازه‌ای و توجه متقاطع}
\label{sec:v5-strategy}

بر پایۀ \lr{V4}، نسخۀ \lr{V5} \emph{نوآوری معماری} در پردازش ویژگی و محاسبۀ نمونه‌های اولیه را بررسی می‌کند:

\begin{enumerate}
\item \textbf{توجه درگاهی به ویژگی (\lr{GFA})}: با الهام از \lr{TabNet}~\cite{arik2021tabnet}، \lr{V5} مکانیزم انتخاب ویژگی تطبیقی به‌ازای نمونه معرفی می‌کند. به‌جای یکسان‌رفتار با همه‌ی ویژگی‌ها، درگاه توجه چندگامه یاد می‌گیرد کدام ویژگی‌ها برای هر بیمار مرتبط‌ترند. در هر یک از $S=3$ گام توجه، ماسک نرم ویژگی $\mathbf{a}_s = \text{\lr{softmax}}(\mathbf{W}_s \mathbf{h} \odot \mathbf{p}_s)$ محاسبه می‌شود که $\mathbf{p}_s$ پیش‌مقیاس با $\gamma - \mathbf{a}_{s-1}$ کاهش یافته است و ویژگی‌های متفاوت در هر گام تشویق می‌شوند. بازنمایی‌های ویژگی ماسک‌شده در گام‌ها تجمیع و از راه درگاه آموزش‌دیده با مسیر ترنسفورمر می‌آمیزند.

\item \textbf{سر نمونه‌اولیه \lr{Cross-Attention}}: به‌جای مقایسۀ پرس‌وجوها با میانگین نمونه‌های اولیه از طریق شباهت کسینوسی (شبکۀ نمونه‌اولیه استاندارد)، \lr{V5} از لایۀ \lr{Cross-Attention} چندسر استفاده می‌کند که هر نمونۀ پرس‌وجو به‌طور هم‌زمان به همه‌ی نمونه‌های اولیه توجه می‌کند. این اجازه می‌دهد مدل توابع شباهت \emph{غیرخطی، وابسته به زمینه} بین پرس‌وجوها و بازنمایی‌های کلاس یاد بگیرد و ساختار روابط غنی‌تر از معیارهای فاصله ساده بگیرد.

\item \textbf{منظم‌سازی سازگاری}: در آموزش، پایداری پیش‌بینی تحت اختلال‌های کوچک ورودی ($\sigma = 0.02$) با جریمه \lr{KL} اجبار می‌شود: $\mathcal{L}_{\text{\lr{consist}}} = \text{\lr{KL}}(p_\theta(\mathbf{x}) \| p_\theta(\mathbf{x} + \boldsymbol{\epsilon}))$. این به‌عنوان پیش‌زمینۀ همواری روی مرز تصمیم عمل می‌کند.

\item \textbf{مهندسی ویژگی گسترده‌شده}: فضای ویژگی از \pnum{92} به \pnum{132} بعد با تعاملات بالینی سه‌جانبه (ترکیب کراتینین، لاکتات، و \lr{BUN})، شاخص‌های چارک، و نسبت‌های ویژۀ حوزه (\lr{BUN}/کراتینین، \lr{AST}/\lr{ALT}، \lr{shock index}) گسترش یافت.

\item \textbf{تجمیع پنج‌بذری}: افزایش از \pnum{3} به \pnum{5} بذر واریانس تجمیع را کاهش و تاب‌آوری را بهبود می‌دهد.
\end{enumerate}

نتایج: تجمیع کم‌نمونه \lr{V5} به \lr{AUROC} \textbf{\pnum{0.722}} \cnum{[0.669, 0.772]} با \lr{AUROC} به‌ازای بذر از \pnum{0.703} تا \textbf{\pnum{0.737}} می‌رسد؛ بالاترین \lr{AUROC} تکی کم‌نمونه مشاهده‌شده در همه‌ی آزمایش‌ها. فرا یادگیر انباشتی \lr{V5} به \lr{AUROC} \textbf{\pnum{0.730}} \cnum{[0.680, 0.778]}، \lr{F1} \pnum{0.661}، و \lr{MCC} \pnum{0.333} می‌رسد. ماژول \lr{GFA} الگوهای توجه ویژگی معنادار بالینی یاد می‌گیرد: کراتینین، لاکتات، و هموگلوبین وزن‌های توجه بالا و پایدار دریافت می‌کنند، سازگار با رتبه‌بندی اهمیت ویژگی \lr{Random Forest}.

\subsection{راهبرد \pnum{8}: مدل کم‌نمونه \lr{V6} با تجمیع عمیق و انباشت \lr{XGBoost}}
\label{sec:v6-strategy}

\lr{V6} می‌آزماید آیا \emph{تنوع معماری} در \lr{Ensemble} می‌تواند با ترکیب رمزگذارهای ساختاراً متفاوت عملکرد را بیشتر بهبود دهد:

\begin{enumerate}
\item \textbf{تجمیع معماری ناهمگن}: به‌جای آموزش همان معماری با بذرهای متفاوت، \lr{V6} سه معماری بنیاداً متفاوت آموزش می‌دهد: (الف)~\textbf{\lr{FT-Transformer}}؛ بردارهای نهفتۀ آموزش‌دیده به‌ازای ویژگی با تجمیع توجه، (ب)~\textbf{\lr{GFA-Transformer}}؛ رمزگذار دومسیره که توجه درگاهی به ویژگی را با ترنسفورمر ویژگی گروه‌بندی‌شده می‌آمیزد، و (ج)~\textbf{\lr{MLP-Mixer}}؛ معماری بدون توجه با \lr{MLP}های مخلوط‌کنندۀ توکن و کانال. هر معماری با \pnum{2} بذر آموزش داده می‌شود (\pnum{6} مدل).

\item \textbf{انباشت \lr{XGBoost}}: فرا یادگیر از رگرسیون لجستیک به درخت‌های تقویت‌شدۀ گرادیان (\lr{XGBoost}) ارتقا می‌یابد که روابط غیرخطی بین پیش‌بینی‌های مدل پایه را می‌گیرد. احتمالات \lr{Ensemble} به‌ازای هر معماری به‌عنوان ویژگی‌های انباشت جداگانه در کنار پیش‌بینی‌های \lr{RF} و مؤلفه‌های \lr{PCA} نهفته‌سازی قرار می‌گیرند.

\item \textbf{پیش‌بینی مطابق (\lr{Conformal prediction})}: پیش‌بینی مطابق تقسیم‌شده مجموعه‌های پیش‌بینی معتبر با تضمین پوشش ($1 - \alpha = 0.90$) فراهم می‌کند. آستانۀ کالیبراسیون مطابق $\hat{q} = 0.304$ میانگین اندازۀ مجموعۀ پیش‌بینی \pnum{1.40} می‌دهد، یعنی \pnum{40}\% بیماران هر دو کلاس را در مجموعۀ پیش‌بینی دریافت می‌کنند (موارد نامطمئن)، در حالی که \pnum{60}\% پیش‌بینی تک‌کلاسه با پوشش حاشیه‌ای \pnum{90}\% دارند.
\end{enumerate}

نتایج: تجمیع ناهمگن \lr{V6} به \lr{AUROC} \textbf{\pnum{0.725}} \cnum{[0.676, 0.773]} می‌رسد. نتایج به‌ازای معماری تفاوت قابل توجه نشان می‌دهد: \lr{FT-Transformer} (\pnum{0.707})، \lr{GFA-Transformer} (\pnum{0.692})، و \lr{MLP-Mixer} (\pnum{0.642}). عملکرد ضعیف‌تر \lr{MLP-Mixer}، یعنی \pnum{0.065} \lr{AUROC} پایین‌تر از \lr{FT-Transformer}، نشان می‌دهد مکانیزم‌های خودتوجه برای یادگیری تعامل بین‌ویژگی در دادۀ جدولی بالینی نقش مهمی دارند؛ مخلوط‌کردن توکن به‌تنهایی کافی نیست. مدل انباشتی \lr{XGBoost} \lr{V6} (\lr{AUROC} \pnum{0.689}) از فرا یادگیر لجستیک \lr{V4} (\pnum{0.732}) به‌دلیل بیش‌برازش فرا یادگیر تقویت‌شدۀ گرادیان روی مجموعۀ اعتبارسنجی کوچک ($n=320$) عقب می‌ماند.

\textbf{بینش کلیدی}: تنوع معماری به‌طور خودکار عملکرد تجمیع را بهبود نمی‌دهد؛ ضعیف‌ترین معماری (\lr{MLP-Mixer}) تجمیع را بیش از آن‌که ناهمگنی کمک کند تخریب می‌کند. رویکرد \lr{V4}، \lr{V5} یعنی تجمیع چندبذری هم‌معماری با انباشت لجستیک کالیبره‌شده برای این مقیاس داده بهینه می‌ماند.

\subsection{تحلیل بهبودها}

بهبودهای تدریجی چند یافته‌ی کلیدی را نشان می‌دهند:

\begin{enumerate}
\item \textbf{پیش‌آموزش \lr{SupCon} دگرگون‌کننده است}: بزرگ‌ترین سهم در بهبود \lr{V2} پیش‌آموزش مقابله‌ای نظارت‌شده (راهبرد~\pnum{4}) است. با بهینه‌سازی صریح برای ساختار نهفته‌سازی خوشه‌شده، \lr{SupCon} دقیقاً هندسۀ بازنمایی را می‌سازد که شبکه‌های نمونه‌اولیه نیاز دارند. این در تضاد با پیش‌آموزش آنتروپی متقاطع استاندارد (راهبرد~\pnum{1}) است که تنها بازنمایی‌های کمی بهتر تولید می‌کند.

\item \textbf{مهندسی ویژگی ظرفیت رمزگذار را تقویت می‌کند}: گسترش فضای ویژگی از \pnum{22} به \pnum{55} بعد از طریق جملات تعاملی، نسبت‌ها، و ویژگی‌های چندجمله‌ای سیگنال غنی‌تری به رمزگذار می‌دهد. ترنسفورمر \lr{ETHOS} سپس می‌تواند به تعاملات بین‌ویژگی که در ویژگی‌های خام ضمنی‌اند توجه کند.

\item \textbf{استنتاج ترابردی بهبود قابل توجه می‌دهد}: استفاده از آمار مجموعۀ پرس‌وجو برای پالایش نمونه‌های اولیه در زمان آزمون (\lr{soft k-means}، \pnum{10} تکرار) مدل را به توزیع محلی آزمون سازگار می‌کند و \lr{AUROC} را حدود \prange{0.02}{0.03} نسبت به ارزیابی غیرترابردی بهبود می‌دهد.

\item \textbf{آموزش چندفازی ضروری است}: آموزش سه‌فازی (\lr{SupCon} $\to$ نظارت‌شده $\to$ اپیزودیک) از هر فاز تکی بهتر عمل می‌کند. هر فاز بر قبلی بنا می‌شود: \lr{SupCon} ساختار اولیه می‌سازد، آموزش نظارت‌شده مرزهای تصمیم را صیقل می‌دهد، و آموزش اپیزودیک مدل را با پروتکل ارزیابی کم‌نمونه سازگار می‌کند.

\item \textbf{تقطیر دانش تأثیرگذارترین نوآوری تکی است}: \lr{V3} با تقطیر دانش \lr{RF} به \lr{AUROC} برابر \pnum{0.712} می‌رسد و نسبت به \lr{V2} با مقدار \pnum{0.689}، بهبود \pnum{0.023} دارد. \lr{KD} مرزهای تصمیم \lr{RF} را به‌خوبی به فضای نهفته رمزگذار منتقل می‌کند.

\item \textbf{فاصله تا مدل‌های کلاسیک به‌شدت کم می‌شود}: مدل کم‌نمونه \lr{V3} (\lr{AUROC} \pnum{0.712}) فاصله تا بهترین مدل کلاسیک (\lr{RF} \pnum{0.760}) را به تنها \pnum{0.048} \lr{AUROC} کاهش می‌دهد (معادل کاهش \pnum{66}\% از فاصلۀ اولیه \pnum{0.141}). بازه‌های اطمینان به‌طور قابل‌توجهی هم‌پوشانی دارند.

\item \textbf{انباشت در کمپین \lr{V} فاصله را کاهش می‌دهد}: فرا یادگیر انباشتی \lr{V4} با \lr{AUROC} برابر \pnum{0.732} بخش زیادی از فاصله تا سطر \lr{RF} در جدول~\ref{tab:improved_strategies} را می‌بندد. برای ادعاهای اصلی نسخه‌ی ارسال، باید به اجرای انباشتی منجمد \textbf{\pnum{0.746}} در برابر \textbf{\pnum{0.748}} استناد شود و در صورت نیاز، آزمون‌های جفتی دوباره محاسبه شوند.

\item \textbf{\lr{Proto-MAML} سازگاری زمان آزمون را ممکن می‌کند}: با وجود آن‌که \lr{Proto-MAML} (گام‌های گرادیان زمان آزمون روی پشتیبان) برآورد نقطه‌ای را نسبت به \lr{V3} به‌طور محسوس بهبود نمی‌دهد (\pnum{0.711} در برابر \pnum{0.712})، قابلیت حیاتی برای استقرار بالینی فراهم می‌کند: مدل می‌تواند در زمان استنتاج بدون آموزش مجدد به توزیع بیماران مؤسسه‌ویژه \emph{سازگار} شود. این به‌ویژه در تنظیمات فدرال ارزشمند است که هر بیمارستان جمعیت‌شناسی متفاوتی دارد.

\item \textbf{توجه درگاهی به ویژگی بالاترین \lr{AUROC} تکی کم‌نمونه را ایجاد می‌کند}: \lr{V5} با \lr{GFA} و نمونه‌های \lr{Cross-Attention} بهترین مدل کم‌نمونه تکی را با بذر \pnum{55} و \lr{AUROC} برابر \pnum{0.737} تولید می‌کند. با این حال، میانگین تجمیع پنج‌بذری (\pnum{0.722}) پایین‌تر از انباشتی \lr{V4} (\pnum{0.732}) است و نشان می‌دهد \lr{GFA} به مقداردهی اولیه حساس‌تر است.

\item \textbf{تنوع معماری در این مقیاس داده کمک نمی‌کند}: تجمیع ناهمگن \lr{V6} (\pnum{0.725}) از تجمیع‌های هم‌معماری \lr{V4}، \lr{V5} عقب می‌ماند. معماری \lr{MLP-Mixer} (\lr{AUROC} \pnum{0.642}) فاقد مکانیزم خودتوجه حیاتی برای تعامل بین‌ویژگی است و تجمیع را پایین می‌کشد. این یافته با ادبیات اخیر یادگیری عمیق جدولی هم‌راستا است که معماری‌های مبتنی بر توجه را بر \lr{MLP} در دادۀ ساخت‌یافته برتر می‌داند.

\item \textbf{انباشت لجستیک کالیبره‌شده از انباشت \lr{XGBoost} بهتر است}: روی مجموعۀ اعتبارسنجی کوچک ($n=320$)، فرا یادگیر تقویت‌شدۀ گرادیان در \lr{V6} (\lr{AUROC} \pnum{0.689}) بیش‌برازش می‌کند، در حالی که رگرسیون لجستیک با کالیبراسیون ایزوتونیک در \lr{V4} (\lr{AUROC} \pnum{0.732}) خوب تعمیم می‌یابد. این نتیجه نشان می‌دهد برای فرا یادگیرهای آموزش‌دیده روی مجموعه‌های اعتبارسنجی کوچک، مدل‌های خطی منظم‌شده بر روش‌های \lr{Ensemble} غیرخطی ترجیح دارند.
\end{enumerate}

با وجود این بهبودها، تفاوت‌های باقی‌ماندۀ عملکرد را می‌توان به موارد زیر نسبت داد:

\begin{enumerate}
\item \textbf{سربار بازنمایی}: بازنمایی توکن‌های علائم از طریق ترنسفورمر پیچیدگی افزوده دارد، در حالی که مدل‌های درخت‌بنیاد مستقیماً فضای ویژگی را افراز می‌کنند. برای دادۀ جدولی متوسط، این سربار تا حدی سود بازنمایی‌های آموزش‌دیده را خنثی می‌کند.
\item \textbf{واریانس ارزیابی اپیزودیک}: حتی با میانگین‌گیری \lr{Ensemble} و پالایش ترابردی، پروتکل ارزیابی اپیزودیک واریانسی معرفی می‌کند که مدل‌های نظارت‌شده از آن اجتناب می‌کنند.
\item \textbf{مقیاس داده}: با \pnum{1,280} نمونۀ آموزش، رمزگذار دادۀ محدودی برای یادگیری بازنمایی‌های قابل‌تعمیم دارد. \lr{SupCon} با بهره‌وری بیشتر از برچسب‌های موجود این را تخفیف می‌دهد اما نمی‌تواند کاملاً جبران کند.
\end{enumerate}

\textbf{جمع‌بندی (هم‌تراز با جداول اولیه):} اجرای انباشتی یکپارچه به \lr{AUROC} \textbf{\pnum{0.746}} می‌رسد، یعنی حدود \pnum{0.002} پایین‌تر از برآورد نقطه‌ای جدولی \lr{SOTA} (\textbf{\pnum{0.748}}). روایت \lr{V1--V6} و جدول~\ref{tab:improved_strategies} همچنان به‌عنوان نقشه‌راه \emph{روش‌شناختی} ارزشمند است؛ اما ادعاهای چکیده و خلاصۀ اجرایی باید ابتدا به جداول اولیه استناد کنند.
