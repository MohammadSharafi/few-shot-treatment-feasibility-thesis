\chapter{پیشینه و کارهای مرتبط}
\label{ch:background}

این فصل مبانی نظری و کارهای پیشین زیربنای رویکرد ما را مرور می‌کند. به یادگیری کم‌نمونه، یادگیری فدرال، بازنمایی دادۀ بالینی، و مجموعۀ دادۀ \lr{MIMIC\text{-}IV} می‌پردازیم. سازمان فصل چنین است: نخست یادگیری کم‌نمونه (تاریخچه، فرمول‌بندی مسئله، شبکه‌های نمونه‌ای و تطبیقی، \lr{MAML}، آموزش اپیزودیک)، سپس یادگیری فدرال (انگیزه، \lr{FedAvg}، حریم خصوصی، دادۀ غیر \lr{IID})، بازنمایی دادۀ بالینی (دادۀ خام، نرمال‌سازی، توکن‌ها)، \lr{MIMIC\text{-}IV} (مرور کلی، طرحواره، کوهورت، برچسب‌ها، گره‌های بیماری)، کارهای مرتبط در پیش‌بینی بالینی، ترنسفورمرها در دادۀ بالینی، یادگیری نامتوازن، تفسیرپذیری، و مرور جامع ادبیات. هر بخش پیش‌زمینۀ لازم برای فهم روش و نتایج ما را فراهم می‌کند.

\section{یادگیری کم‌نمونه}
%==============================================================================

\subsection{تاریخچه و تحول}

حوزۀ یادگیری کم‌نمونه از پژوهش‌های فرا یادگیری در دهۀ \pnum{2010} برخاسته است. کارهای اولیه دربارۀ «یادگیری برای یادگیری» بررسی می‌کردند که چگونه الگوریتم‌ها می‌توانند با تجربه در وظایف گوناگون بهبود یابند. انقلاب \lr{ImageNet} نشان داد شبکه‌های عمیق می‌توانند بازنمایی‌های نیرومند از مجموعه‌های بزرگ بیاموزند؛ یادگیری کم‌نمونه پرسید آیا بازنمایی‌های مشابه را می‌توان با تعداد اندک نمونه و بهره‌گیری از ساختار بین وظایف آموخت.

\subsection{فرمول‌بندی مسئله}

یادگیری کم‌نمونه به چالش تعمیم از تعداد اندک نمونۀ برچسب‌خورده به‌ازای هر کلاس می‌پردازد~\cite{snell2017prototypical}. در تنظیم استاندارد $N$-راه $K$-شات، $N$ کلاس و $K$ نمونۀ پشتیبان به‌ازای هر کلاس داریم. هدف طبقه‌بندی نمونه‌های پرس‌وجو تنها با مجموعۀ پشتیبان است، بدون آموزش اضافی.

به‌صورت رسمی، $\mathcal{S} = \{(x_i, y_i)\}_{i=1}^{N \cdot K}$ را مجموعۀ پشتیبان و $\mathcal{Q} = \{(x_j, y_j)\}_{j=1}^{M}$ را مجموعۀ پرس‌وجو می‌نامیم. طبقه‌بند $\hat{y} = f(x; \mathcal{S})$ می‌خواهیم که با توجه به $\mathcal{S}$ برچسب $x$ را پیش‌بینی کند. کلید، یادگیری بازنمایی و معیار فاصله است که بین وظایف تعمیم یابد.

\subsection{شبکه‌های نمونه‌ای}

شبکه‌های نمونه‌ای~\cite{snell2017prototypical} برای هر کلاس یک نمونه‌ای (\lr{prototype}) به‌صورت میانگین بردارهای نهفتۀ پشتیبان محاسبه می‌کنند:
\begin{equation}
\mathbf{c}_k = \frac{1}{|\mathcal{S}_k|} \sum_{(x_i, y_i) \in \mathcal{S}_k} f_\theta(x_i)
\end{equation}
که در آن $\mathcal{S}_k = \{(x_i, y_i) : y_i = k\}$ و $f_\theta$ شبکۀ نهفته‌سازی است. پرس‌وجوی $x$ با نزدیک‌ترین نمونه‌ای در فضای نهفته طبقه‌بندی می‌شود:
\begin{equation}
p(y = k \mid x) = \frac{\exp(-d(f_\theta(x), \mathbf{c}_k) / \tau)}{\sum_{k'} \exp(-d(f_\theta(x), \mathbf{c}_{k'}) / \tau)}
\end{equation}
که در آن $d$ فاصلۀ اقلیدسی و $\tau$ پارامتر دما است. در پیاده‌سازی ما از شباهت کسینوسی استفاده می‌شود.

\subsection{شبکه‌های تطبیقی}

شبکه‌های تطبیقی~\cite{vinyals2016matching} با توجه روی مجموعۀ پشتیبان برچسب پرس‌وجو را محاسبه می‌کنند:
\begin{equation}
p(y \mid x, \mathcal{S}) = \sum_{i} a(x, x_i) \, \mathbb{1}[y_i = y]
\end{equation}
که در آن $a(x, x_i)$ وزن توجه است. این اجازه می‌دهد نمونه‌های پشتیبان مختلف بر اساس شباهت به پرس‌وجو سهم متفاوتی داشته باشند.

\subsection{شبکه‌های رابطه‌ای}

شبکه‌های رابطه‌ای~\cite{sung2018learning} معیار فاصلۀ ثابت (اقلیدسی، کسینوسی) را با یک \emph{ماژول رابطه آموخته‌شده} جایگزین می‌کنند. برای بردار نهفتۀ پرس‌وجوی $f_\theta(x)$ و نمونه‌ای کلاس $\mathbf{c}_k$, امتیاز رابطه چنین است:
\begin{equation}
r_k = g_\phi(f_\theta(x), \mathbf{c}_k)
\end{equation}
که در آن $g_\phi: \mathbb{R}^{2d} \to [0,1]$ یک شبکۀ عصبی (معمولاً \lr{MLP} کوچک با خروجی سیگموئید) است که تابع شباهت وابسته به وظیفه را می‌آموزد. مدل به‌صورت انتها-به-انتها با تابع زیان میانگین مربعات خطا آموزش داده می‌شود:
\begin{equation}
\mathcal{L} = \sum_{(x,y) \in \mathcal{Q}} \sum_k (r_k - \mathbb{1}[y=k])^2
\end{equation}

مزیت شبکه‌های رابطه‌ای بر شبکه‌های نمونه‌ای انعطاف است: ماژول رابطه می‌تواند توابع شباهت غیرخطی یاد بگیرد که مرزهای تصمیم پیچیده را دربرگیرد. عیب، هزینۀ محاسباتی بیشتر و خطر بیش‌برازش با دادۀ فرا-آموزش محدود است. برای امکان‌پذیری درمان، جایی که مرز تصمیم ممکن است به تعاملات غیرخطی ویژگی (مثلاً تعامل کراتینین و سن) وابسته باشد، شبکه‌های رابطه‌ای جایگزین نظری جذابی برای طبقه‌بندی مبتنی بر نمونه‌ای فراهم می‌کنند.

شبکه‌های رابطه‌ای را به‌عنوان نقطۀ مقایسه در مطالعۀ حذف تدریجی پیاده‌سازی می‌کنیم. ماژول رابطه از دو لایۀ تمام‌اتصال با فعال‌سازی \lr{ReLU} تشکیل شده است:
\begin{equation}
g_\phi(\mathbf{u}, \mathbf{v}) = \sigma(W_2 \cdot \text{\lr{ReLU}}(W_1 \cdot [\mathbf{u}; \mathbf{v}; \mathbf{u} \odot \mathbf{v}; |\mathbf{u} - \mathbf{v}|] + b_1) + b_2)
\end{equation}
که در آن $[\cdot;\cdot]$ الحاق، $\odot$ ضرب درایه‌به‌درایه، و $\sigma$ تابع سیگموئید است. ورودی شامل چهار ویژگی تعاملی است: دو نهفته‌سازی، حاصل‌ضرب درایه‌به‌درایه‌شان (گرفتن تعاملات ضربی)، و قدر مطلق تفاضل‌شان (گرفتن فاصله).

\subsection{شبکه‌های سیامی}

شبکه‌های سیامی~\cite{koch2015siamese} با پردازش جفت ورودی از شاخه‌های یکسان (اشتراک وزن) تابع شباهت می‌آموزند:
\begin{equation}
s(x_1, x_2) = \sigma(\|f_\theta(x_1) - f_\theta(x_2)\|_1)
\end{equation}
که در آن $\sigma$ تابع سیگموئید است. شبکه با زیان متضاد روی جفت‌های هم‌کلاس (مثبت) و غیرهم‌کلاس (منفی) آموزش داده می‌شود. اگرچه برای یادگیری کم‌نمونه بنیادین است، شبکه‌های سیامی به مقایسه‌های جفتی ($\mathcal{O}(NK)$ به‌ازای هر پرس‌وجو) نیاز دارند و از شبکه‌های نمونه‌ای ($\mathcal{O}(N)$ پس از محاسبۀ نمونه‌ای) کم‌کارآمدترند. شبکه‌های سیامی را پیاده‌سازی نمی‌کنیم اما اهمیت تاریخی آن‌ها را یادآور می‌شویم.

\subsection{فرا یادگیری مستقل از مدل (\lr{MAML})}

\lr{MAML}~\cite{finn2017model} مقداردهی اولیه $\theta$ را چنان می‌آموزد که چند گام گرادیان روی وظیفۀ جدید عملکرد خوب بدهد. هدف فرا-یادگیری چنین است:
\begin{equation}
\min_\theta \sum_{\mathcal{T}_i} \mathcal{L}_{\mathcal{T}_i}(f_{\theta_i'})
\end{equation}
که در آن $\theta_i' = \theta - \alpha \nabla_\theta \mathcal{L}_{\mathcal{T}_i}(f_\theta)$ به‌روزرسانی وابسته به وظیفه است. \lr{MAML} انعطاف‌پذیرتر است اما پرهزینه؛ برای کارایی از شبکه‌های نمونه‌ای استفاده می‌کنیم.

\lr{MAML} به محاسبۀ گرادیان مرتبۀ دوم (گرادیان گرادیان‌ها) نیاز دارد که حافظه و محاسبه را نسبت به روش‌های مرتبۀ اول حدود \textbf{\pnum{2} تا \pnum{3} برابر} افزایش می‌دهد. تقریب‌های مرتبۀ اول (\lr{FOMAML}، \lr{Reptile}~\cite{nichol2018reptile}) با حذف جملۀ مرتبۀ دوم این هزینه را از بین می‌برند:
\begin{equation}
\theta \gets \theta - \beta \frac{1}{|\mathcal{T}|} \sum_{\mathcal{T}_i} (\theta - \theta_i')
\end{equation}
\lr{Reptile} جهت از پارامترهای اولیه به پارامترهای سازگاریافته با وظیفه را میانگین می‌گیرد و جایگزینی ساده و مقیاس‌پذیر فراهم می‌کند. برای مسئلۀ ما (طبقه‌بندی دوجمله‌ای با وظایف محدود)، سربار محاسباتی \lr{MAML} ممنوع‌کننده نیست، اما تنوع محدود اپیزودهای فرا-آموزش (\pnum{5} گرۀ بیماری) شاید پیچیدگی اضافی را توجیه نکند. از این رو شبکه‌های نمونه‌ای را به‌عنوان روش اصلی کم‌نمونه به کار می‌بریم و \lr{MAML} را برای کار آتی با مجموعه‌های فرا-آموزش بزرگ‌تر ذکر می‌کنیم.

\subsection{آموزش اپیزودیک}

آموزش اپیزودیک در طول فرا-آموزش وظایف کم‌نمونه را شبیه‌سازی می‌کند. هر اپیزود وظیفه $\mathcal{T}_i$ با پشتیبان $\mathcal{S}_i$ و پرس‌وجوی $\mathcal{Q}_i$ نمونه‌برداری می‌کند. مدل طوری آموزش داده می‌شود که زیان روی $\mathcal{Q}_i$ با توجه به $\mathcal{S}_i$ کمینه شود. این با سناریوی استنتاج هم‌خوان است و تعمیم را بهبود می‌بخشد.

\subsection{تضمین‌های نظری}

تحت فرض‌های معین، مانند داده‌ی حاصل از مخلوط گاوسی و کلاس‌های جدا، می‌توان نشان داد شبکه‌های نمونه‌ای با $K \to \infty$ به طبقه‌بند بهینۀ بیزی همگرا می‌شوند. در عمل، $K$ متناهی است و توزیع‌های غیرگاوسی عملکرد را محدود می‌کنند. در یادگیری متریک، فضای نهفته باید نمونه‌های مشابه را نزدیک و نمونه‌های ناهمگون را دور نگه دارد. دما $\tau$ نیز تیزی توزیع \lr{softmax} را کنترل می‌کند؛ مقدار بزرگ‌تر \(\tau\) پیش‌بینی‌های نرم‌تری ایجاد می‌کند.

\subsection{کاربردها در سلامت}

یادگیری کم‌نمونه به طبقه‌بندی تصویر پزشکی (مثلاً رادیوگرافی، هیستوپاتولوژی)، پیش‌بینی پاسخ دارو، و پیش‌بینی پیامد بالینی به کار رفته است. بیشتر کار بر تصویربرداری تمرکز دارد، جایی که ستون‌فقرات‌های کانولوشنی و انتقال از \lr{ImageNet} بازنمایی‌های نیرومند می‌دهند. دادۀ بالینی جدولی؛ آزمایشگاه، علائم حیاتی، جمعیت‌شناسی؛ در تنظیم کم‌نمونه کم‌کاوش مانده است. چالش‌ها عبارت‌اند از: (\pnum{1}) نبود ساختار فضایی طبیعی برای کانولوشن؛ (\pnum{2}) انواع و واحدهای ناهمگون ویژگی؛ (\pnum{3}) فقدان داده و نمونه‌برداری نامنظم؛ (\pnum{4}) مجموعه‌های داده کوچک‌تر از بنچمارک‌های بینایی. این کار پارادایم را به امکان‌پذیری درمان از آزمایشگاه و علائم حیاتی با بازنمایی مبتنی بر توکن و رمزگذار ترنسفورمر گسترش می‌دهد.

\section{یادگیری فدرال}
%==============================================================================

\subsection{انگیزه}

دادۀ سلامت در بیمارستان‌ها، کلینیک‌ها و مناطق توزیع شده است. تجمیع مرکزی داده مقررات حریم خصوصی را نقض می‌کند و ریسک امنیتی دارد. یادگیری فدرال مدل‌ها را روی دادۀ توزیع‌شده بدون جابه‌جایی نمونه‌های خام به سرور مرکزی آموزش می‌دهد~\cite{mcmahan2017communication}.

\subsection{میانگین‌گیری فدرال (\lr{FedAvg})}

\lr{FedAvg} به‌روزرسانی‌های مدل محلی را تجمیع می‌کند. فرض کنید $K$ مشتری مجموعه‌های داده $\mathcal{D}_1, \ldots, \mathcal{D}_K$ را نگه دارند. در دور $t$:
\begin{enumerate}
\item سرور مدل سراسری $\theta_t$ را به مشتریان می‌فرستد.
\item هر مشتری $k$ به‌روزرسانی محلی محاسبه می‌کند: $\theta_{t+1}^{(k)} = \theta_t - \eta \nabla \mathcal{L}_k(\theta_t)$.
\item سرور تجمیع می‌کند: $\theta_{t+1} = \sum_k \frac{n_k}{n} \theta_{t+1}^{(k)}$.
\end{enumerate}
که در آن $n_k = |\mathcal{D}_k|$ و $n = \sum_k n_k$.

\subsection{حریم خصوصی و امنیت}

یادگیری فدرال نمایش داده را کاهش می‌دهد: تنها به‌روزرسانی‌های مدل (نه دادۀ خام) منتقل می‌شوند. می‌توان حریم تفاضلی با برش گرادیان و افزودن نویز افزود. تجمیع امن می‌تواند به‌روزرسانی‌های فردی را از سرور پنهان کند. در این پایان‌نامه از \lr{FedAvg} استاندارد بدون حریم تفاضلی استفاده می‌شود.

\subsection{همگرایی و ارتباطات}

\lr{FedAvg} تحت فرض‌های استاندارد (گرادیان‌های کران‌دار، زیان لیپشیتز) همگرا می‌شود. تعداد دورهای ارتباطی لازم به ناهمگونی داده بستگی دارد: دادۀ ناهمگون‌تر ممکن است به دورهای بیشتر یا روش‌های تطبیقی (مثلاً \lr{FedProx}، \lr{SCAFFOLD}) نیاز داشته باشد. از \pnum{20} دور استفاده می‌کنیم؛ در آزمایش‌های ما \lr{AUROC} اعتبارسنجی تا حدود دور \prange{15}{20} به سکون می‌رسد.

\subsection{\lr{FedProx}: مدیریت ناهمگونی}

\lr{FedProx}~\cite{li2020federated} با افزودن جملۀ تنظیم‌کنندۀ مجاورتی به هدف محلی هر مشتری، چالش دادۀ ناهمگون (غیر \lr{IID}) بین مشتریان را برطرف می‌کند:
\begin{equation}
\min_\theta \mathcal{L}_k(\theta) + \frac{\mu}{2} \|\theta - \theta_t\|^2
\end{equation}
که در آن $\theta_t$ مدل سراسری جاری و $\mu > 0$ قدرت جملۀ مجاورتی را کنترل می‌کند. این از انحراف بیش‌ازحد مشتریان فردی از مدل سراسری در طول آموزش محلی جلوگیری می‌کند؛ به‌ویژه وقتی توزیع‌های دادۀ محلی به‌طور محسوس متفاوت‌اند (مانند پارتیشن‌بندی مبتنی بر بیماری ما). جملۀ مجاورتی مانند تنظیم‌کنندۀ ضمنی عمل می‌کند که سازگاری محلی و سازگاری سراسری را متعادل می‌کند.

در تنظیم ما، پنج گرۀ بیماری (سپسیس، قلبی، تنفسی، کلیوی، دیابت) به‌طور طبیعی توزیع‌های ویژگی و نسبت کلاس متفاوتی دارند. برای مثال بیماران کلیوی ممکن است به‌طور سیستماتیک کراتینین بالاتری از بیماران قلبی داشته باشند و جابه‌جایی توزیع بین گره‌ها ایجاد شود. \lr{FedProx} با محدودکردن به‌روزرسانی‌های محلی این امر را تخفیف می‌دهد. \lr{FedAvg} را به‌عنوان خط پایۀ اصلی به کار می‌بریم و \lr{FedProx} را برای کار آتی ذکر می‌کنیم، به‌ویژه اگر عملکرد با ترکیب ناهمگون‌تر گره‌ها افت کند.

\subsection{\lr{SCAFFOLD}: کاهش واریانس}

\lr{SCAFFOLD}~\cite{karimireddy2020scaffold} با متغیرهای کنترلی برای اصلاح انحراف مشتری به کار می‌رود. هر مشتری متغیر کنترلی $c_k$ را نگه می‌دارد که تفاضل گرادیان بین اهداف محلی و سراسری را تخمین می‌زند:
\begin{equation}
\theta_{t+1}^{(k)} = \theta_t - \eta (\nabla \mathcal{L}_k(\theta_t) - c_k + c)
\end{equation}
که در آن $c = \frac{1}{N}\sum_k c_k$ متغیر کنترلی سراسری است. \lr{SCAFFOLD} نسبت به \lr{FedAvg} و \lr{FedProx}، به‌ویژه تحت ناهمگونی بالای داده، همگرایی سریع‌تری با کاهش واریانس به‌روزرسانی‌های گرادیان به دست می‌آورد. هزینه، ارتباط اضافی است (هر دور هم به‌روزرسانی مدل و هم به‌روزرسانی متغیر کنترلی منتقل می‌شود). برای تنظیم \pnum{5} گره‌ای ما با ناهمگونی متوسط، \lr{SCAFFOLD} ممکن است بهبود حاشیه‌ای بدهد؛ برای آزمایش‌های چندسایتی آتی که ناهمگونی انتظار می‌رود بالاتر باشد در نظر گرفته می‌شود.

\subsection{کارایی ارتباطات}

هزینۀ ارتباطات گلوگاه عملی در یادگیری فدرال است. چند راهکار برای کاهش پهنای باند پیشنهاد شده است:

\begin{enumerate}
\item \textbf{فشرده‌سازی گرادیان}: پراکندسازی \lr{top-$k$} تنها $k$ مؤلفۀ بزرگ‌ترین گرادیان را منتقل می‌کند~\cite{alistarh2017qsgd}. با $k = 1\%$ پارامترها، نسبت فشرده‌سازی حدود \textbf{\pnum{100} برابر} با افت دقت اندک قابل دستیابی است.
\item \textbf{کوانتیزاسیون گرادیان}: نمایش مقادیر گرادیان با بیت کمتر (مثلاً \pnum{8} بیت یا \pnum{1} بیت). \lr{SignSGD}~\cite{bernstein2018signsgd} تنها علامت هر مؤلفۀ گرادیان را می‌فرستد.
\item \lr{Local SGD}: افزایش دورهای آموزش محلی بین دورهای ارتباطی. از \pnum{3} دور محلی به‌ازای هر دور استفاده می‌کنیم؛ افزایش آن ارتباطات را کم می‌کند اما ممکن است انحراف مشتری را بیشتر کند.
\item \textbf{به‌روزرسانی جزئی مدل}: انتقال تنها لایه‌های به‌روزشده (مثلاً تنها سر طبقه‌بندی با منجمدکردن رمزگذار). در معماری ما رمزگذار (حدود \pnum{1.8} میلیون پارامتر) غالب است؛ انتقال تنها \pnum{0.2} میلیون پارامتر سر پهنای باند را حدود \pnum{90}\% کاهش می‌دهد.
\end{enumerate}

در آزمایش‌ها از \lr{FedAvg} استاندارد با انتقال کامل مدل استفاده می‌شود. با \pnum{5} گره و \pnum{20} دور، کل ارتباطات حدود \LTRformula{5 \times 20 \times 2 \approx 200} انتقال مدل است (هر کدام حدود \pnum{8} مگابایت برای \pnum{2} میلیون پارامتر \lr{float32})، جمع حدود \pnum{1.6} گیگابایت. برای پژوهش قابل‌تحمل است اما برای استقرار بلادرنگ روی شبکه‌های محدود ممکن است به بهینه‌سازی نیاز باشد.

\subsection{دادۀ غیر \lr{IID}}

دادۀ فدرال اغلب غیر \lr{IID} است: هر گره ممکن است توزیع کلاس یا ویژگی متفاوتی داشته باشد. این می‌تواند به همگرایی آسیب بزند. \lr{MIMIC\text{-}IV} را بر اساس گروه بیماری (\lr{ICD-10}) پارتیشن می‌کنیم که گره‌های غیر \lr{IID} طبیعی می‌دهد. درجه ناهمگونی را می‌توان با فاصلۀ جابه‌جایی زمین (\lr{EMD}) بین توزیع‌های برچسب محلی و سراسری، یا واگرایی \lr{KL} بین توزیع‌های ویژگی محلی و سراسری کمی‌سازی کرد. در پارتیشن ما نرخ امکان‌پذیری از \pnum{35}\% (گرۀ کلیوی) تا \pnum{48}\% (گرۀ تنفسی) متغیر است و ناهمگونی متوسط برچسب ایجاد می‌کند.

دادۀ غیر \lr{IID} یک تنش ایجاد می‌کند: مدل‌های محلی بیش‌برازش به توزیع‌های محلی، در حالی که مدل سراسری باید تعمیم دهد. \lr{FedAvg} با میانگین‌وزنی که به‌صورت ضمنی به مرکز وزن‌داده‌شده منظم می‌کند، این را مدیریت می‌کند. رویکردهای پیچیده‌تر (\lr{FedProx}، \lr{SCAFFOLD}، یادگیری فدرال شخصی‌سازی‌شده~\cite{smith2017federated}) وقتی ناهمگونی شدید است جایگزین می‌شوند. آزمایش‌ها نشان می‌دهد برای تنظیم \pnum{5} گره‌ای ما \lr{FedAvg} کفایت می‌کند و افت عملکرد ناچیز نسبت به آموزش متمرکز است.

\subsection{کاربردها در سلامت}

یادگیری فدرال به پیش‌بینی مبتنی بر \lr{EHR}، تصویربرداری پزشکی، و کارآزمایی‌های بالینی به کار رفته است. پروژه‌های برجستۀ \lr{FL} در سلامت شامل:
\begin{itemize}
\item \textbf{همکاری \lr{Intel--Penn Medicine}}: بخش‌بندی تومور مغزی فدرال در \pnum{30} مؤسسه، نشان داد که \lr{FL} به \pnum{99}\% عملکرد متمرکز می‌رسد~\cite{sheller2020federated}.
\item \textbf{\lr{NVIDIA Clara}}: بستر یادگیری فدرال برای تصویربرداری پزشکی در چند شبکۀ بیمارستانی.
\item \textbf{\lr{HealthChain}}: پروژۀ مالی‌شده توسط اتحادیۀ اروپا برای تحلیل فدرال کارآزمایی بالینی در بیمارستان‌های اروپایی.
\item \textbf{\lr{FedHealth}}: یادگیری انتقالی فدرال برای پایش سلامت پوشیدنی~\cite{chen2020fedhealth}.
\end{itemize}
این کار یادگیری فدرال را با فرا یادگیری کم‌نمونه برای امکان‌پذیری درمان ترکیب می‌کند؛ ترکیبی که در کاربردهای پیشین \lr{FL} سلامت کمتر کاوش شده است.

\subsection{دادۀ بالینی خام}

دادۀ \lr{ICU} معمولاً شامل موارد زیر است:
\begin{itemize}
\item \textbf{مقادیر آزمایشگاهی}: کراتینین، پتاسیم، سدیم، گلوکز، هموگلوبین، پلاکت، \lr{WBC}، لاکتات و غیره.
\item \textbf{علائم حیاتی}: ضربان قلب، فشار خون، دما، \lr{SpO$_2$}, نرخ تنفس.
\item \textbf{جمعیت‌شناسی}: سن، جنس، نوع پذیرش.
\item \textbf{پیامدها}: محل ترخیص، مدت بستری، مرگ‌ومیر، بستری مجدد.
\end{itemize}

مقادیر خام ناهمگون‌اند: واحدها، دامنه‌ها، و الگوهای فقدان داده متفاوت. نرمال‌سازی و جایگزینی بحرانی است.

\subsection{راهبردهای نرمال‌سازی}

رویکردهای رایج:
\begin{enumerate}
\item \textbf{مقیاس \lr{min-max}}: $x' = (x - x_{\min}) / (x_{\max} - x_{\min})$. پیش از مقیاس‌بندی به دامنۀ بالینی برش می‌دهیم.
\item \textbf{\lr{Z-score}}: $x' = (x - \mu) / \sigma$. به داده‌های پرت حساس است.
\item \textbf{مبتنی بر رتبه}: نگاشت به صدک‌ها. مقاوم اما اطلاعات مقیاس را از دست می‌دهد.
\end{enumerate}
از مقیاس \lr{min-max} با برش دامنۀ بالینی به $[0,1]$ استفاده می‌شود.

\subsection{بازنمایی‌های مبتنی بر توکن}

ترنسفورمرها انتظار دنباله‌ای از توکن دارند. در \lr{NLP} توکن‌ها قطعات کلمه‌اند. برای دادۀ بالینی \emph{توکن‌های علائم} (\lr{Symptom Tokens}) تعریف می‌کنیم: هر توکن \lr{[symptom\_id, intensity, time\_delta, modality]} را رمزگذاری می‌کند. این اجازه می‌دهد مدل روی اندازه‌گیری‌های مختلف و روابطشان توجه کند.

\subsection{جدولی در برابر دنباله‌ای}

مدل‌های جدولی (مثلاً \lr{Random Forest}، \lr{XGBoost}) بردارهای ویژگی با طول ثابت به کار می‌برند. مدل‌های دنباله‌ای (مثلاً ترنسفورمرها) دنباله‌های توکن با طول متغیر. هر دو پشتیبانی می‌شوند: ویژگی‌های جدولی برای خطوط پایه، دنباله‌های توکن برای \lr{ETHOS}.

\subsection{مرور کلی}

\lr{MIMIC\text{-}IV v3.1}~\cite{johnson2023mimic} پایگاه دادۀ بزرگ و آزاد \lr{ICU} از مرکز پزشکی \lr{Beth Israel Deaconess} (\lr{BIDMC}) است. جایگزین \lr{MIMIC\text{-}III}~\cite{johnson2016mimic} با ساختار دادۀ به‌روز و پوشش گسترده‌تر است. پایگاه داده دادۀ ناشناس‌شده برای بیش از \pnum{250,000} اقامت \lr{ICU} دارد، شامل اندازه‌گیری‌های آزمایشگاهی (از \texttt{\lr{labevents}})، علائم حیاتی و دادۀ نمودارشده (از \texttt{\lr{chartevents}})، تشخیص‌های \lr{ICD-9} و \lr{ICD-10}، اعمال جراحی، و پیامدها (محل ترخیص، مرگ‌ومیر، مدت بستری). دسترسی تکمیل فرآیند اعتبارسنجی (\lr{CITI}، توافق استفاده از داده) از طریق \lr{PhysioNet} را می‌طلبد. کوهورت \pnum{2,000} بیمار ما نمونۀ لایه‌ای است که برای آزمایش‌های بازتولیدپذیر طراحی شده و از نظر محاسباتی قابل‌مدیریت مانده است.

\subsection{ساختار داده}

جداول کلیدی و نقش آن‌ها:
\begin{itemize}
\item \texttt{\lr{patients}}: جمعیت‌شناسی (سن، جنس، سال لنگر). برای فیلتر سن و تعریف کوهورت به کار می‌رود.
\item \texttt{\lr{admissions}}: نوع پذیرش (انتخابی، اورژانسی)، محل ترخیص، زمان پذیرش و ترخیص. برای تعریف برچسب امکان‌پذیری بحرانی است.
\item \texttt{\lr{icustays}}: شناسه‌های اقامت \lr{ICU} (\texttt{\lr{stay\_id}})، \texttt{\lr{subject\_id}}، \texttt{\lr{hadm\_id}}، \texttt{\lr{intime}}، \texttt{\lr{outtime}}، مدت بستری. جدول اصلی برای استخراج کوهورت.
\item \texttt{\lr{labevents}}: اندازه‌گیری‌های آزمایشگاهی با \texttt{\lr{itemid}} (از \texttt{\lr{d\_labitems}})، \texttt{\lr{valuenum}}، \texttt{\lr{charttime}}. \pnum{15} نوع آزمایش (کراتینین، پتاسیم، سدیم و غیره) از \pnum{24} ساعت اول استخراج می‌شود.
\item \texttt{\lr{chartevents}}: علائم حیاتی و دادۀ نمودارشده‌ی دیگر. \pnum{7} علامت حیاتی (ضربان قلب، فشار خون، دما، \lr{SpO2}، نرخ تنفس) از \texttt{\lr{d\_items}} استخراج می‌شود.
\item \texttt{\lr{diagnoses\_icd}}: تشخیص‌های \lr{ICD-9}/\lr{ICD-10}. برای تخصیص تشخیص اولیه و پارتیشن‌بندی گرۀ بیماری به کار می‌رود.
\end{itemize}
اتصال‌ها از طریق \texttt{\lr{subject\_id}}، \texttt{\lr{hadm\_id}}، و \texttt{\lr{stay\_id}} انجام می‌شود. برای پرس‌وجوهای \lr{SQL} به پیوست~\ref{app:sql} مراجعه کنید.

\subsection{تعریف کوهورت}

کوهورت \pnum{2,000} بیمار با این شرایط استخراج می‌شود:
\begin{itemize}
\item تنها نخستین اقامت \lr{ICU}.
\item حداقل \pnum{4} ساعت مدت بستری.
\item سن $\geq$ \pnum{18}.
\item \lr{ICD-10} اولیه برای تخصیص گرۀ بیماری در دسترس باشد.
\end{itemize}

\subsection{تعریف برچسب}

امکان‌پذیری درمان دوجمله‌ای است: \pnum{1} اگر محل ترخیص در $\{$\lr{HOME}، \lr{HOME HEALTH CARE}، \lr{HOME WITH HOME IV PROVIDR}، \lr{REHAB/DISTINCT PART HOSP}، \lr{REHAB}$\}$ ظرف \pnum{30} روز از اقامت \lr{ICU} باشد و بستری مجدد در \pnum{30} روز نباشد؛ در غیر این صورت \pnum{0}. این با تعاریف بالینی مقصدهای ترخیص «ممکن» هم‌خوان است.

\subsection{گره‌های بیماری}

بر اساس \lr{ICD-10} اولیه به پنج گره پارتیشن می‌کنیم:
\begin{itemize}
\item گره \pnum{1} (سپسیس): \lr{A40}، \lr{A41}، \lr{R652}.
\item گره \pnum{2} (قلبی): \lr{I21}، \lr{I22}، \lr{I25}، \lr{I50}، \lr{I47}، \lr{I48}، \lr{I49}.
\item گره \pnum{3} (تنفسی): \lr{J18}، \lr{J44}، \lr{J96}، \lr{J80}، \lr{J15}.
\item گره \pnum{4} (کلیوی): \lr{N17}، \lr{N18}، \lr{N19}.
\item گره \pnum{5} (دیابت): \lr{E10}، \lr{E11}، \lr{E13}.
\end{itemize}
این آموزش فدرال با دادۀ محلی وابسته به بیماری را ممکن می‌سازد.

\section{کارهای مرتبط: پیش‌بینی بالینی}

\subsection{پیش‌بینی مبتنی بر \lr{EHR}}

پرونده‌های سلامت الکترونیک (\lr{EHR}) برای پیش‌بینی مرگ‌ومیر، ریسک بستری مجدد، و پیش‌بینی مدت بستری به کار رفته‌اند. رویکردهای کلاسیک رگرسیون لجستیک، جنگل تصادفی، یا تقویت گرادیان روی ویژگی‌های دست‌ساز را به کار می‌برند. روش‌های یادگیری عمیق شامل شبکه‌های بازگشتی برای دنباله‌های زمانی و ترنسفورمرها برای دادۀ ساخت‌یافته است. این کار با تمرکز بر تنظیمات کم‌نمونه و فدرال متفاوت است.

\subsection{پیش‌بینی ترخیص}

پیش‌بینی مقصد ترخیص مسئلۀ پرکار است. عوامل شامل وضعیت عملکردی، همراه بیماری، حمایت اجتماعی، و مسیر اولیۀ \lr{ICU} است. بیشتر کارهای پیشین دادۀ متمرکز و یادگیری نظارت‌شدۀ استاندارد به کار می‌برند. بررسی می‌کنیم آیا فرا یادگیری کم‌نمونه می‌تواند عملکرد کلاسیک را وقتی داده محدود و توزیع‌شده است برابر کند یا به آن نزدیک شود.

\subsection{یادگیری عمیق در برابر جدولی}

روی دادۀ جدولی، تقویت گرادیان (\lr{XGBoost}، \lr{LightGBM}، \lr{CatBoost}) اغلب یادگیری عمیق را پیشی می‌گیرد. شبکه‌های عصبی ممکن است به دادۀ بیشتر یا طراحی دقیق معماری نیاز داشته باشند. نتایج منجمد \lr{Exp1} ما هم‌راستاست: \lr{Random Forest} (\lr{AUROC} \pnum{0.762}) از \texttt{\lr{Proposed\_FewShot}} (\pnum{0.611}) بهتر است. این شکاف نشان می‌دهد برای این وظیفه و اندازۀ مجموعه، وقتی آموزش متمرکز ممکن است، مدل‌های جدولی همچنان انتخاب ترجیحی‌اند.

\subsection{یادگیری کم‌نمونه روی دادۀ جدولی}

یادگیری کم‌نمونه برای طبقه‌بندی تصویر (\lr{miniImageNet}، \lr{CUB}، \lr{Omniglot}) و پردازش زبان طبیعی به‌گستردگی مطالعه شده، اما کاربرد به دادۀ جدولی کم‌کاوش مانده است. دادۀ جدولی چالش‌های ویژه برای فرا یادگیری دارد:

\begin{enumerate}
\item \textbf{ناهمگونی ویژگی}: برخلاف تصاویر (که همه پیکسل‌ها فضای بازنمایی مشترک دارند)، ویژگی‌های جدولی انواع (پیوسته، طبقه‌ای، دوجمله‌ای) و مقیاس‌های متنوع (مثلاً کراتینین به \lr{mg/dL} در برابر ضربان قلب به \lr{bpm}) دارند. طرح‌های توکن‌سازی باید این تنوع را پوشش دهند.
\item \textbf{تعاملات ویژگی}: مدل‌های مبتنی بر درخت به‌طور طبیعی تعاملات را از طریق انشعاب می‌گیرند؛ شبکه‌های نمونه‌ای به فضای نهفته برای رمزگذاری این تعاملات تکیه می‌کنند که ممکن است به دادۀ آموزش فراوان نیاز داشته باشد.
\item \textbf{رژیم نمونۀ کم}: وقتی کل مجموعه کوچک است (مثلاً \pnum{2,000} بیمار)، مجموعۀ فرا-آموزش ممکن است تنوع کافی برای یادگیری بردارهای نهفتۀ تعمیم‌یافته بین اپیزودها نداشته باشد.
\item \textbf{فقدان ساختار فضایی}: تصاویر همبستگی مکانی محلی دارند؛ ویژگی‌های جدولی این سوگیری القایی را ندارند. ترنسفورمرهای اعمال‌شده به دادۀ جدولی باید روابط ویژگی را از صفر بیاموزند.
\end{enumerate}

کار اخیر \lr{Iwata} و \lr{Kumagai} (\pnum{2020}) فرا یادگیری را به رگرسیون جدولی اعمال کرد، و \lr{Hegselmann} و همکاران (\pnum{2023}) \lr{TabLLM} را برای طبقه‌بندی جدولی کم‌نمونه با بردارهای نهفتۀ مدل زبانی کاوش کردند. این رویکردها پیشنهاد می‌کنند یادگیری جدولی مؤثر کم‌نمونه ممکن است نیاز به (الف) مجموعه‌های پیش‌آموزش بسیار بزرگ، (ب) توکن‌سازی وابسته به وظیفه، یا (ج) معماری‌های ترکیبی که بازنمایی‌های آموخته‌شده را با ویژگی‌های کلاسیک یکی کند، داشته باشد. معماری \texttt{\lr{ETHOS\_RF\_Hybrid}} ما با جهت (ج) هم‌راستا است.

\subsection{مکانیزم‌های توجه برای دادۀ بالینی}

خود-توجه~\cite{vaswani2017attention} به هر توکن اجازه می‌دهد به همه توکن‌های دیگر توجه کند و وابستگی‌های دوربرد را بگیرد. در دادۀ بالینی، مدل می‌تواند روابط بین آزمایشگاه و علائم حیاتی بیاموزد (مثلاً کراتینین و \lr{BUN} برای عملکرد کلیوی؛ لاکتات و فشار خون برای وضع همودینامیک). رمزگذار \lr{ETHOS} اصلاحات دامنه‌ویژه اضافه می‌کند: (\pnum{1}) ماتریس سوگیری هم‌وقوعی قابل‌آموزش که دانش پیشین هم‌وقوعی علائم را رمزگذاری می‌کند؛ (\pnum{2}) وزن‌دهی شدت که توجه را به بزرگی مقادیر بالینی مقیاس می‌کند؛ (\pnum{3}) فروپاشی زمانی که نفوذ اندازه‌گیری‌های قدیمی‌تر را کاهش می‌دهد. این اصلاحات \lr{ETHOS} را از ترنسفورمرهای عمومی متمایز می‌کند و بخشی از نوآوری رویکرد ماست.

\subsection{جزئیات معماری \lr{ETHOS}}

رمزگذار \lr{ETHOS} (\lr{Efficient Transformer for Health Outcome prediction using Symptom tokens})~\cite{sharafi2023ethos} معماری ترنسفورمر استاندارد را با سه مکانیزم دامنه‌ویژه برای دادۀ بالینی گسترش می‌دهد:

\paragraph{سوگیری هم‌وقوعی.} ماتریس سوگیری قابل‌آموزش $B \in \mathbb{R}^{V \times V}$ (که $V$ اندازۀ واژگان علامت است) به \lr{logit}های توجه پیش از \lr{softmax} افزوده می‌شود:
\begin{equation}
\text{\lr{Attention}}(Q, K, V) = \text{\lr{softmax}}\left(\frac{QK^\top}{\sqrt{d_k}} + B_{\text{\lr{co-occur}}}\right) V
\end{equation}
درایه $B_{ij}$ قدرت هم‌وقوعی پیشین بین علامت $i$ و علامت $j$ را رمزگذاری می‌کند. می‌توان از دانش بالینی (مثلاً هم‌وقوعی کراتینین و \lr{BUN} در پانل کلیوی) مقداردهی اولیه کرد یا از داده آموخت. سوگیری مدل را تشویق می‌کند به اندازه‌گیری‌های بالینی مرتبط توجه کند و سوگیری القایی فقدان ساختار فضایی در دادۀ جدولی را جبران می‌کند.

\paragraph{وزن‌دهی شدت.} توجه استاندارد همه توکن‌ها را بدون توجه به بزرگی بالینی یکسان می‌پندارد. \lr{ETHOS} توجه را با شدت (مقدار نرمال‌شده) هر توکن مدوله می‌کند:
\begin{equation}
\alpha_{ij}^{\text{\lr{intensity}}} = \alpha_{ij} \cdot \sigma(w_I \cdot \text{\lr{intensity}}_j + b_I)
\end{equation}
که در آن $\alpha_{ij}$ وزن توجه استاندارد است و $\sigma(w_I \cdot \text{\lr{intensity}}_j + b_I)$ تابع دروازه‌ای آموخته‌شده از شدت توکن است. مقادیر افراطی (کراتینین بسیار بالا، پلاکت بسیار پایین) توجه تقویت‌شده می‌گیرند؛ منعکس‌کنندۀ عمل بالینی که مقادیر غیرعادی اهمیت تشخیصی نامتناسب دارند.

\paragraph{فروپاشی زمانی.} وقتی چند اندازه‌گیری در زمان‌های مختلف موجود است، \lr{ETHOS} فروپاشی نمایی زمانی برای کاهش نفوذ مشاهدات قدیمی‌تر اعمال می‌کند:
\begin{equation}
w_{\text{\lr{time}}}(t) = \exp(-\lambda \cdot |t - t_{\text{\lr{ref}}}|)
\end{equation}
که در آن $\lambda$ نرخ فروپاشی قابل‌آموزش و $t_{\text{\lr{ref}}}$ زمان مرجع (معمولاً تازه‌ترین مشاهده یا زمان پیش‌بینی) است. این شهود بالینی را منعکس می‌کند: کراتینین \pnum{2} ساعت پیش از \pnum{20} ساعت پیش اطلاعاتی‌تر است. در پیاده‌سازی ما با دادۀ تجمیع‌شدۀ \pnum{24} ساعت اول، اثر فروپاشی زمانی محدود است (همه اندازه‌گیری‌ها از همان پنجره‌اند)، اما مکانیزم برای تعمیم به ورودی‌های سری زمانی گنجانده شده است.

\paragraph{مقایسه با ترنسفورمر عمومی.} جدول~\ref{tab:ethos_vs_generic} \lr{ETHOS} را با رمزگذار ترنسفورمر عمومی در انتخاب‌های طراحی کلیدی مقایسه می‌کند.

\subsection{رویکردهای توکن‌سازی علامت در ادبیات}

مفهوم توکن‌سازی اندازه‌گیری‌های بالینی برای پردازش ترنسفورمر از چند جهت پژوهشی برخاسته است:

\begin{enumerate}
\item \textbf{نسخه‌های بالینی \lr{BERT}.} مدل‌های \lr{ClinicalBERT} \cite{alsentzer2019publicly} و \lr{BioBERT} \cite{lee2020biobert} متن بالینی (یادداشت‌ها و گزارش‌ها) را با توکن‌سازی \lr{wordpiece} پردازش می‌کنند. این مدل‌ها برای زبان طبیعی طراحی شده‌اند، نه برای اندازه‌گیری‌های ساخت‌یافته.
\item \textbf{نهفته‌سازی کدهای پزشکی.} مدل‌های \lr{BEHRT} \cite{li2020behrt} و \lr{Med-BERT} \cite{rasmy2021med} کدهای تشخیصی (\lr{ICD-9}/\lr{ICD-10}) را به‌عنوان توکن در معماری شبیه \lr{BERT} نهفته‌سازی می‌کنند. این خانواده کدهای طبقه‌ای را پوشش می‌دهد، نه مقادیر پیوسته‌ی آزمایشگاه.
\item \textbf{\lr{TabTransformer}}~\cite{huang2020tabtransformer}: ویژگی‌های طبقه‌ای را به‌عنوان توکن نهفته‌سازی می‌کند و توجه ترنسفورمر اعمال می‌کند، در حالی که ویژگی‌های عددی را به‌صورت اسکالر پس از ترنسفورمر الحاق می‌کند. رویکرد ما با توکن‌سازی \emph{هم} اطلاعات طبقه‌ای (\lr{ID} علامت) و \emph{هم} عددی (شدت) در یک توکن یکپارچه متفاوت است.
\item \textbf{\lr{SAINT}}~\cite{somepalli2021saint}: توجه بین-نمونه و بین-ویژگی برای دادۀ جدولی اعمال می‌کند. برخلاف رویکرد ما، \lr{SAINT} بازنمایی مبتنی بر توکن به کار نمی‌برد بلکه هر ویژگی را بعد توجه می‌پندارد.
\item \textbf{\lr{FT-Transformer}}~\cite{gorishniy2024revisiting}: همه ویژگی‌ها (عددی و طبقه‌ای) را به بردار نهفته تبدیل می‌کند و لایه‌های ترنسفورمر استاندارد اعمال می‌کند. به رویکرد ما نزدیک‌تر است اما اصلاحات دامنه‌ویژه بالینی (سوگیری هم‌وقوعی، وزن‌دهی شدت، فروپاشی زمانی) ندارد.
\end{enumerate}

طرح توکن‌های علائم ما با این موارد متمایز است: (\pnum{1}) نرمال‌سازی دامنۀ بالینی با کران‌های برش دامنه‌ویژه؛ (\pnum{2}) رمزگذاری صریح مودالیتی (آزمایش در برابر علامت حیاتی)؛ (\pnum{3}) ساختار چهاربعدی توکن برای هویت، بزرگی، زمان، و نوع؛ و (\pnum{4}) سازگاری با آموزش اپیزودیک کم‌نمونه از طریق دنباله‌های با طول ثابت.

مقالات کلیدی که کار ما را روشن می‌کنند را خلاصه می‌کنیم. \textbf{\lr{Snell et al. (2017)}} \lr{MAML} را برای فرا یادگیری معرفی کرد؛ به‌خاطر پیچیدگی محاسباتی از نمونه‌ای استفاده می‌کنیم. \textbf{\lr{Johnson et al. (2023)}} \lr{MIMIC\text{-}IV} را منتشر کرد؛ از طرحواره \lr{v3.1} استفاده می‌کنیم. \textbf{\lr{Sharafi \& Aminian (2023)}} \lr{ETHOS} را برای پیش‌بینی پیامد سلامت پیشنهاد کرد؛ برای یادگیری کم‌نمونه سازگارش می‌کنیم. \textbf{\lr{Lin et al. (2017)}} زیان کانونی (\lr{focal loss}) را برای دادۀ نامتوازن پیشنهاد کرد؛ برای سر کم‌نمونه به کار می‌بریم. \textbf{\lr{Lundberg \& Lee (2017)}} \lr{SHAP} را برای تفسیرپذیری معرفی کرد؛ برای توضیح مدل استفاده می‌کنیم. این مراجع بنیان نظری و تجربی رویکرد ما را فراهم می‌کنند.

\subsection{مبانی فرا یادگیری}

فرا یادگیری، یا «یادگیری برای یادگیری»، ریشه در علوم شناختی و پژوهش اولیۀ \lr{AI} دارد. بینش کلیدی این است که خود یادگیری می‌تواند بهینه شود: به‌جای آموزش مدل از صفر روی هر وظیفۀ جدید، می‌توان مقداردهی اولیه، پیش‌زمینه، یا الگوریتم یادگیری آموخت که بین وظایف تعمیم یابد. این وقتی وظایف ساختار مشترک دارند (مثلاً همه طبقه‌بندی فنوتیپ‌های بالینی) اما هر وظیفه دادۀ برچسب‌خورده محدود دارد، به‌ویژه مرتبط است.

\subsection{یادگیری انتقالی و سازگاری دامنه}

یادگیری انتقالی مدلی را که روی دامنۀ مبدأ پیش‌آموزش دیده روی دامنۀ هدف تنظیم می‌کند. یادگیری کم‌نمونه را می‌توان به‌عنوان انتقال افراطی دید: دامنۀ هدف تعداد بسیار اندک نمونۀ برچسب‌خورده دارد. سازگاری دامنه جابه‌جایی توزیع بین مبدأ و هدف را پوشش می‌دهد؛ در تنظیم ما، گره‌های بیماری مختلف ممکن است توزیع‌های متفاوتی داشته باشند. یادگیری فدرال قید می‌کند که دادۀ مبدأ و هدف را نمی‌توان تجمیع کرد.

\subsection{یادگیری متضاد و یادگیری متریک}

یادگیری متریک بردارهای نهفتۀی می‌آموزد که نمونه‌های مشابه نزدیک و نمونه‌های ناهمگون دور باشند. شبکه‌های نمونه‌ای به‌صورت ضمنی یادگیری متریک به کار می‌برند: فضای نهفته طوری آموزش داده می‌شود که نمونه‌ای‌های کلاس جدا شوند. یادگیری متضاد (مثلاً \lr{SimCLR}، \lr{MoCo}) به تصویربرداری پزشکی اعمال شده؛ گسترش به دادۀ بالینی جدولی جهتی باز است.

\subsection{چالش‌های ویژۀ سلامت}

یادگیری ماشین در سلامت با چالش‌های ویژه روبه‌روست: (\pnum{1}) مقرراتی (\lr{FDA}، علامت‌گذاری \lr{CE})، (\pnum{2}) تفسیرپذیری (بالین‌دانان باید پیش‌بینی را بفهمند)، (\pnum{3}) انصاف (اجتناب از سوگیری علیه زیرگروه‌ها)، (\pnum{4}) اعتبار زمانی (مدل‌ها ممکن است با تکامل عمل ضعیف شوند). این کار این موارد را عمیق پوشش نمی‌دهد اما خط پایه‌ای برای پژوهش آتی فراهم می‌کند.

%==============================================================================
\section{ترنسفورمرها در دادۀ بالینی}
%==============================================================================

\subsection{مکانیزم‌های توجه}

معماری ترنسفورمر~\cite{vaswani2017attention,devlin2019bert} \lr{NLP} را دگرگون کرد و برای دادۀ بالینی سازگار شده است. خود-توجه به مدل اجازه می‌دهد هنگام محاسبۀ بازنمایی‌ها، وزن‌های مختلف به توکن‌های ورودی (مثلاً مقادیر آزمایشگاه، علائم حیاتی) بدهد. برای دادۀ بالینی جدولی، رمزگذارهای مبتنی بر توکن مانند \lr{ETHOS} توجه روی توکن‌های علائم را برای گرفتن تعاملات اعمال می‌کنند.

\subsection{رمزگذاری موقعیتی}

برخلاف متن دنباله‌ای، توکن‌های بالینی ممکن است ترتیب ضمنی داشته باشند (مثلاً زمان اندازه‌گیری). از رمزگذاری موقعیتی برای تزریق اطلاعات ترتیب استفاده می‌شود. رمزگذاری‌های سینوسی و آموخته‌شده رایج‌اند؛ از فرمول‌بندی استاندارد ترنسفورمر پیروی می‌کنیم.

\subsection{ترنسفورمرهای بالینی}

\lr{BEHRT}~\cite{li2020behrt} و سایر نسخه‌های \lr{BERT} بالینی روی دنباله‌های \lr{EHR} پیش‌آموزش می‌بینند. \lr{G-BERT}~\cite{shang2020gbert} ساختار گراف را می‌گنجاند. \lr{TabTransformer}~\cite{huang2020tabtransformer} و \lr{SAINT}~\cite{somepalli2021saint} ترنسفورمر را به دادۀ جدولی اعمال می‌کنند. رمزگذار \lr{ETHOS} ما سبک‌تر است و برای آموزش اپیزودیک کم‌نمونه طراحی شده، نه پیش‌آموزش مقیاس بزرگ.

\section{یادگیری در شرایط نامتوازن}
%==============================================================================

\subsection{نامتوازنی کلاس در دادۀ بالینی}

کوهورت‌های \lr{ICU} اغلب پیامدهای نامتوازن دارند: مثلاً \pnum{42}\% ممکن در برابر \pnum{58}\% ناممکن در کوهورت ما. آنتروپی متقاطع استاندارد ممکن است به کلاس اکثریت سوگیری داشته باشد. از زیان کانونی~\cite{lin2017focal} برای کم‌وزن‌کردن نمونه‌های آسان و وزن کلاس برای خطوط پایه استفاده می‌کنیم. برای ارزیابی، \lr{AUROC}~\cite{hanley1982meaning} و امتیاز \lr{F1} کلان استانداردند؛ ضریب همبستگی \lr{Matthews}~\cite{matthews1975comparison} نیز گزارش می‌شود.

\subsection{\lr{SMOTE} و نمونه‌برداری مجدد}

\lr{SMOTE} با درون‌یابی بین نزدیک‌ترین همسایه‌ها نمونۀ اقلیت می‌سازد. در حذف تدریجی اختیاری \lr{SMOTE} اعمال می‌شود؛ برای وظیفۀ ما وزن‌دهی کلاس و تنظیم آستانه کفایت می‌کند.

\subsection{معیارهای ارزیابی}

برای دادۀ نامتوازن، \lr{AUROC} نسبت به آستانۀ تصمیم مستقل است، در حالی که امتیاز \lr{F1} کلان به آستانه وابسته است و میانگین \lr{F1} را به‌ازای هر کلاس گزارش می‌کند. هر دو معیار گزارش می‌شوند؛ \lr{AUROC} معیار اصلی مقایسۀ مدل‌هاست.

\section{تفسیرپذیری در یادگیری ماشین سلامت}

\subsection{اهمیت ویژگی}

مدل‌های مبتنی بر درخت (\lr{Random Forest}، \lr{XGBoost}) اهمیت بومی ویژگی می‌دهند. از اهمیت \lr{Gini} و اهمیت جایگشت استفاده می‌کنیم. \lr{SHAP}~\cite{lundberg2017unified} توضیحات مستقل از مدل ارائه می‌دهد؛ برای تفسیرپذیری محلی به کار می‌بریم. \lr{LIME}~\cite{ribeiro2016should} جایگزین است؛ برای سازگاری با مدل‌های مبتنی بر درخت از \lr{SHAP} استفاده می‌کنیم.

\subsection{تصویرسازی توجه}

برای مدل‌های ترنسفورمر، وزن‌های توجه نشان می‌دهند مدل به کدام توکن‌ها توجه می‌کند. در فصل~\ref{ch:results}، توجه را از طریق انتساب ویژگی \lr{SHAP} برای پیش‌بینی‌های فردی تحلیل می‌کنیم.

\subsection{اعتبارسنجی بالینی}

تفسیرپذیری باید با دانش بالینی هم‌راستا باشد. ویژگی‌های برتر (کراتینین، گلوکز، هموگلوبین) با تعیین‌کننده‌های شناخته‌شدۀ ترخیص هم‌خوان‌اند. تفسیر بالینی در فصل~\ref{ch:results} بحث می‌شود.

\section{مرور جامع ادبیات}
%==============================================================================

\subsection{تحول و بنچمارک‌های یادگیری کم‌نمونه}

یادگیری کم‌نمونه از شبکه‌های سیامی اولیه \cite{koch2015siamese} و شبکه‌های تطبیقی \cite{vinyals2016matching} به رویکردهای نمونه‌ای مدرن \cite{snell2017prototypical} و فرا یادگیری \cite{finn2017model,ravi2017optimization} تکامل یافته است. این حوزه به‌گستردگی مرور شده است \cite{wang2020generalizing}. بنچمارک‌های کلیدی آن شامل \lr{Omniglot} \cite{lake2015human}، \lr{mini-ImageNet} و \lr{Meta-Dataset} \cite{triantafillou2020meta} است. کارهای اخیر نیز پرسیده‌اند آیا خطوط پایۀ ساده \cite{dhillon2019baseline} می‌توانند با فرا یادگیرهای پیچیده برابری کنند. این پایان‌نامه همان پرسش را به دادۀ بالینی جدولی گسترش می‌دهد.

\subsection{یادگیری فدرال: سامانه‌ها و نظریه}

سامانه‌های یادگیری فدرال \cite{bonawitz2019towards,he2020fedml} مقیاس، ناهمگونی و حریم خصوصی را مدیریت می‌کنند. تحلیل نظری \cite{karimireddy2020scaffold,reddi2020fedopt} همگرایی غیر \lr{IID} را پوشش می‌دهد. کاربردهای سلامت \cite{rieke2020future,kaissis2021secure,zhang2021federated} نیز امکان‌پذیری این پارادایم را نشان داده‌اند. پیاده‌سازی \lr{FedAvg} ما پروتکل استاندارد \cite{mcmahan2017communication} را دنبال می‌کند؛ در این پایان‌نامه حریم تفاضلی یا تجمیع امن اضافه نمی‌کنیم.

\subsection{پیش‌بینی بالینی: وضعیت هنر}

پیش‌بینی مبتنی بر \lr{EHR} از رگرسیون لجستیک \cite{awad2017random} به یادگیری عمیق \cite{rajkomar2019scalable,choi2016doctor,pham2016deepcare} پیش رفته است. یادگیری چندوظیفه‌ای \cite{harutyunyan2019multitask} کارایی نمونه را بهبود می‌بخشد. مطالعات بنچمارک \cite{purushotham2018benchmarking} نشان می‌دهند تقویت گرادیان اغلب یادگیری عمیق را روی \lr{EHR} جدولی پیشی می‌گیرد. نتایج منجمد ما نیز همین الگو را نشان می‌دهد: \lr{Random Forest} با \lr{AUROC} \pnum{0.762} از مدل پیشنهادی کم‌نمونه با \pnum{0.611} بهتر است و معماری ترکیبی \lr{ETHOS} و \lr{Random Forest} با \pnum{0.717} یک راه‌حل میانی ارائه می‌دهد.

\subsection{یادگیری عمیق جدولی}

کار اخیر شامل \lr{TabNet}~\cite{arik2021tabnet}، \lr{TabTransformer}~\cite{huang2020tabtransformer}، \lr{SAINT}~\cite{somepalli2021saint}، و \lr{FT-Transformer}~\cite{gorishniy2024revisiting} است. \lr{TabNet} توجه دنباله‌ای برای انتخاب ویژگی به کار می‌برد؛ \lr{TabTransformer} خود-توجه را روی بردارهای نهفتۀ ویژگی‌های طبقه‌ای و عددی اعمال می‌کند؛ \lr{SAINT} توجه روی ردیف و ستون دارد. رمزگذار \lr{ETHOS} ما مفهوماً شبیه \lr{TabTransformer} است اما روی توکن‌های علائم (\lr{ID}های گسسته + شدت) به‌جای ستون‌های جدولی خام عمل می‌کند. روی مجموعه‌های بزرگ پیش‌آموزش نمی‌بینیم؛ رمزگذار از صفر برای وظیفۀ کم‌نمونه آموزش داده می‌شود.

\subsection{مجموعه‌های دادۀ بنچمارک در \lr{ML} بالینی}

فراتر از \lr{MIMIC}، بنچمارک‌های مهم شامل \lr{eICU} برای دادۀ چندمرکزه \lr{ICU}، \lr{UK Biobank} برای سلامت جمعیتی و مجموعه‌های اختصاصی \lr{EHR} هستند. جدول~\ref{tab:benchmark_datasets} این مجموعه‌ها را مقایسه می‌کند. \lr{MIMIC\text{-}IV} تمرکز \lr{ICU}، دسترسی آزاد و مستندسازی قوی دارد؛ \lr{eICU} تنوع چندمرکزه اضافه می‌کند؛ و \lr{UK Biobank} دادۀ جمعیتی طولی فراهم می‌سازد. انتخاب \lr{MIMIC\text{-}IV} به سه دلیل انجام شد: هم‌راستایی با مسئلۀ ترخیص \lr{ICU}، دسترس‌پذیری برای پژوهش دانشگاهی و وجود فرایندهای پیش‌پردازش و بنچمارک‌های پذیرفته‌شده در جامعه‌ی پژوهشی.

\subsection{پیش‌بینی ترخیص: نتایج پیشین}

کار پیشین پیش‌بینی ترخیص نتایج متنوع گزارش می‌کند. \lr{Awad et al. (2017)} \lr{Random Forest} و رگرسیون لجستیک روی \lr{MIMIC\text{-}III} برای مقصد ترخیص به \lr{AUROC} \pnum{0.78} رسیدند. \lr{Caruana et al. (2015)} \lr{AUROC} \pnum{0.88} برای پیش‌بینی مرگ‌ومیر روی دادۀ \lr{EHR} گزارش کردند. \lr{Purushotham et al. (2018)} \lr{LSTM} و تقویت گرادیان را روی مرگ‌ومیر \lr{MIMIC\text{-}III} (\lr{AUROC} \pnum{0.85}) بنچمارک کردند. \lr{Rajkomar et al. (2018)} برای پیش‌بینی بستری مجدد \lr{AUROC} \pnum{0.76} به‌دست آوردند. وظیفۀ امکان‌پذیری درمان ما مرتبط اما متمایز است: از دادۀ \pnum{24} ساعت اول، امکان‌پذیری دوجمله‌ای (ترخیص به منزل/توان‌بخشی بدون بستری مجدد) پیش‌بینی می‌کنیم. \lr{AUROC} \pnum{0.76} ما با نتایج پیشین ترخیص و بستری مجدد قابل‌مقایسه است. نوآوری در تنظیم کم‌نمونه و فدرال است، نه فراتر رفتن از بنچمارک‌های متمرکز.

\subsection{\lr{AI} اخلاقی در سلامت}

ملاحظات اخلاقی~\cite{floridi2018ai4people,jobin2019global,gerke2020ethical} و انصاف~\cite{obermeyer2019dissecting,chouldechova2017fairness,rajkomar2019ensuring} برای \lr{AI} سلامت بحرانی است. اصول کلیدی: (\pnum{1}) \textbf{نیک‌کاری}: مدل‌ها باید پیامد بیمار را بهبود دهند؛ هم نتایج مثبت و منفی گزارش می‌کنیم تا اغراق نشود. (\pnum{2}) \textbf{نیک‌نبودن}: اجتناب از آسیب؛ برای استقرار انسان-در-حلقه توصیه می‌کنیم. (\pnum{3}) \textbf{خودمختاری}: بیماران و بالین‌دانان حاکمیت تصمیم را دارند؛ مدل‌ها پشتیبان تصمیم‌اند. (\pnum{4}) \textbf{عدالت}: انصاف بین جمعیت‌شناسی‌ها؛ عملکرد لایه‌بندی‌شده بر اساس سن گزارش می‌شود. (\pnum{5}) \textbf{شفافیت}: تفسیرپذیری از طریق \lr{SHAP} و اهمیت ویژگی. چارچوب‌های مقرراتی (\lr{FDA} برای نرم‌افزار به‌عنوان دستگاه پزشکی، \lr{GDPR} برای اتحادیۀ اروپا) استقرار را حاکم می‌کنند؛ این کار از دادۀ پژوهشی استفاده می‌کند و محصول مقرراتی نیست.

چارچوب‌های مقرراتی (\lr{FDA}، \lr{GDPR}~\cite{voigt2019eu}) استقرار را حاکم می‌کنند. این کار از دادۀ ناشناس‌شدۀ \lr{MIMIC\text{-}IV} استفاده می‌کند؛ اخلاق در پیوست~\ref{app:ethics} بحث می‌شود.

%==============================================================================
\section{جزئیات معماری ترنسفورمر}
%==============================================================================

رمزگذار ترنسفورمر از لایه‌های متناوب چندسر خود-توجه و پیش‌خور جلو تشکیل شده است. هر سر توجه $\text{\lr{head}}_i = \text{\lr{Softmax}}(Q_i K_i^\top / \sqrt{d_k}) V_i$ را محاسبه می‌کند که در آن $Q_i, K_i, V_i$ تصویرهای آموخته‌شده‌اند. خروجی همه سرها الحاق و تصویر می‌شود. نرمال‌سازی لایه پیش و پس از هر زیرلایه اعمال می‌شود (پیکربندی \lr{Pre-LN}). لایۀ پیش‌خور جلو: $\text{\lr{FFN}}(x) = \text{\lr{ReLU}}(xW_1 + b_1)W_2 + b_2$ با $d_{\text{\lr{ff}}} = 512$. \lr{Dropout} (\pnum{0.2}) پس از توجه و \lr{FFN} اعمال می‌شود. \lr{TabNet}، \lr{TabTransformer}، \lr{SAINT}، و \lr{FT-Transformer} ترنسفورمر را به دادۀ جدولی اعمال می‌کنند؛ رمزگذار \lr{ETHOS} ما روی توکن‌های علائم (\lr{ID}های گسسته + شدت) عمل می‌کند و از صفر برای یادگیری کم‌نمونه آموزش داده می‌شود، نه پیش‌آموزش مقیاس بزرگ.

\section{پروتکل ارزیابی و واریانس \lr{K}-شات}
%==============================================================================

برای ارزیابی کم‌نمونه، \pnum{100} اپیزود از مجموعۀ آزمون نمونه‌برداری می‌کنیم. هر اپیزود $K$ پشتیبان و $Q$ پرس‌وجو به‌ازای هر کلاس دارد و میانگین \lr{AUROC} و انحراف معیار بین اپیزودها گزارش می‌شود. برای مدل‌های کلاسیک از کل مجموعۀ آزمون در یک ارزیابی تکی استفاده می‌کنیم. این تفاوت پروتکل توضیح می‌دهد چرا \lr{AUROC} آزمایش \lr{K}-شات در بازۀ \prange{0.557}{0.564} می‌ماند، در حالی که ارزیابی ثابت \lr{Exp1} برای \texttt{\lr{Proposed\_FewShot}} عدد \pnum{0.611} را روی کل مجموعۀ آزمون گزارش می‌کند. کار پیشین روی \lr{MIMIC\text{-}III} \lr{AUROC} \pnum{0.78} برای ترخیص (\lr{Awad et al. 2017}) و \pnum{0.85} برای مرگ‌ومیر (\lr{Purushotham 2018}) گزارش کرد؛ \lr{AUROC} امکان‌پذیری درمان \lr{MIMIC\text{-}IV} ما با خط پایه‌ی کلاسیک در حدود \pnum{0.76} در بازۀ قابل‌مقایسه است.

\section{بنچمارک‌های یادگیری کم‌نمونه}
%==============================================================================

یادگیری کم‌نمونه روی بنچمارک‌هایی مانند \lr{Omniglot}، \lr{mini-ImageNet} و \lr{Meta-Dataset} به‌طور گسترده ارزیابی شده است. این بنچمارک‌ها عمدتاً تصویری‌اند و ساختار فضایی طبیعی دارند؛ در حالی که دادۀ بالینی جدولی چنین ساختاری ندارد و هر ویژگی اسکالری معنای بالینی متفاوتی دارد. این پژوهش پارادایم کم‌نمونه را به پیش‌بینی بالینی جدولی گسترش می‌دهد. تفاوت‌های اصلی عبارت‌اند از: نبود ستون‌فقرات ازپیش‌آموزش‌دیده مانند \lr{ImageNet}، کوچک‌تر بودن مجموعۀ فرا-آموزش (\pnum{2,000} در برابر میلیون‌ها نمونه)، طبقه‌بندی دودویی به‌جای چندکلاسه، و ارزیابی اپیزودیک با $K$ متغیر. \lr{AUROC} متوسط حدود \pnum{0.62} نشان می‌دهد کم‌نمونه جدولی برای رقابت با بنچمارک‌های بینایی احتمالاً به معماری‌های متفاوت یا مجموعه‌های فرا-آموزش بزرگ‌تر نیاز دارد.

\section{یادگیری فدرال: ارتباطات و حریم خصوصی}
%==============================================================================

\lr{FedAvg} به چند دور ارتباط نیاز دارد. در هر دور، سرور مدل سراسری را به مشتریان می‌فرستد، مشتریان مدل را روی دادۀ محلی خود آموزش می‌دهند و به‌روزرسانی‌ها را بازمی‌گردانند. با \pnum{5} گره و \pnum{20} دور، \pnum{100} گام ارتباطی انجام می‌شود و در هر گام مدل کامل، حدود \pnum{2} میلیون پارامتر در معماری ما، منتقل می‌گردد. در استقرار واقعی، فشرده‌سازی گرادیان یا به‌روزرسانی‌های جزئی می‌تواند پهنای باند را کاهش دهد. حریم تفاضلی نیز با برش گرادیان و افزودن نویز گاوسی قابل افزودن است، هرچند در این پایان‌نامه پیاده‌سازی نشده است. تجمیع امن، مثلاً با رمزنگاری هم‌ریختی یا محاسبۀ چندطرفه امن، می‌تواند به‌روزرسانی‌های فردی را از سرور پنهان کند. شکاف عملکرد کوچک، حدود \pnum{0.005} واحد \lr{AUROC} در آزمایش گزارش‌شده، نشان می‌دهد وقتی داده را نمی‌توان تجمیع کرد، یادگیری فدرال برای این وظیفه عملی است.

\section{پیش‌بینی بالینی: فرمول‌بندی وظیفه}
%==============================================================================

امکان‌پذیری درمان وظیفۀ طبقه‌بندی دوجمله‌ای است: با دادۀ اولیۀ \lr{ICU} (\pnum{24} ساعت اول)، پیش‌بینی کنید آیا بیمار به ترخیص ممکن دست می‌یابد. برچسب از محل ترخیص و بستری مجدد \pnum{30} روزه استخراج می‌شود. این فرمول‌بندی محدودیت دارد: (\pnum{1}) مقصد ترخیص ممکن است به عوامل اجتماعی (مثلاً حمایت خانواده، بیمه) وابسته باشد که در آزمایشگاه و علائم حیاتی نیست؛ (\pnum{2}) بستری مجدد \pnum{30} روزه جایگزین بهبودی است؛ (\pnum{3}) برچسب دوجمله‌ای ظرافت را از دست می‌دهد (مثلاً توان‌بخشی در برابر مراقبت پرستاری ماهر). با وجود این محدودیت‌ها، فرمول‌بندی بالینی معنادار و قابل‌مدیریت است. فرمول‌بندی‌های جایگزین شامل: رگرسیون (زمان تا ترخیص)، چندکلاسه (مقصد ترخیص)، یا تحلیل بقا (زمان-تا-رویداد) است. روی طبقه‌بندی دوجمله‌ای برای سادگی و هم‌راستایی با ادبیات پیشین پیش‌بینی ترخیص تمرکز می‌کنیم.

\section{جایگزینی و مدیریت دادۀ گمشده}
%==============================================================================

دادۀ بالینی اغلب مقادیر گمشده دارد. راهبرد زیر به کار می‌رود: (\pnum{1}) برای سری زمانی (آزمایشگاه، علائم حیاتی در \pnum{24} ساعت اول): پرکردن رو به جلو درون بیمار، سپس میانگین جمعیت برای همچنان‌گمشده. (\pnum{2}) برای ویژگی‌های کاملاً گمشده: میانگین جمعیت. (\pnum{3}) بدون پرکردن رو به عقب (تنها دادۀ گذشته نسبت به زمان پیش‌بینی). (\pnum{4}) جایگزینی پیچیده (مثلاً \lr{MICE}، \lr{KNN}) برای اجتناب از نشت داده و هزینۀ محاسباتی به کار نمی‌رود؛ جایگزینی میانگین محافظه‌کارانه است. نرخ فقدان داده به‌ازای ویژگی متفاوت است: آزمایشگاه‌های معمولی (کراتینین، سدیم) \prange{5}{12}\%؛ آزمایشگاه‌های تخصصی (لاکتات، تروپونین) \prange{18}{35}\%؛ علائم حیاتی \prange{2}{8}\%. ویژگی‌ها با بیش از \pnum{30}\% فقدان ممکن است ارزش پیش‌بینی کاهش‌یافته داشته باشند؛ برای کمال نگه داشته می‌شوند و مدل می‌آموزد. در استقرار، دادۀ بلادرنگ ممکن است الگوی فقدان متفاوتی داشته باشد؛ پایش و آموزش مجدد ممکن است لازم شود.

\section{جمع‌بندی فصل}
%==============================================================================

این فصل مبانی نظری کار ما را مرور کرد. به یادگیری کم‌نمونه (شبکه‌های نمونه‌ای، تطبیقی، \lr{MAML}، آموزش اپیزودیک)، یادگیری فدرال (\lr{FedAvg}، حریم خصوصی، دادۀ غیر \lr{IID}، ارتباطات)، بازنمایی دادۀ بالینی (نرمال‌سازی، توکن‌ها، جدولی در برابر دنباله‌ای)، و مجموعۀ دادۀ \lr{MIMIC\text{-}IV} (طرحواره، کوهورت، برچسب‌ها، گره‌های بیماری) پرداختیم. کارهای مرتبط در پیش‌بینی بالینی، ترنسفورمرها برای دادۀ بالینی، یادگیری نامتوازن، تفسیرپذیری، و \lr{AI} اخلاقی را بحث کردیم. همچنین جدول مقایسۀ کارهای پیشین، جزئیات معماری ترنسفورمر، واریانس پروتکل ارزیابی، بنچمارک‌های کم‌نمونه، ارتباطات و حریم خصوصی فدرال، فرمول‌بندی وظیفۀ بالینی، و راهبردهای جایگزینی را ارائه دادیم. فصل بعد روش‌شناسی را به‌تفصیل شرح می‌دهد.
