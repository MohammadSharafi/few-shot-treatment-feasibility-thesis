\section{محدودیت‌ها}
%==============================================================================

\begin{itemize}
\item \textbf{داده}: \lr{MIMIC\text{-}IV} گذشته‌نگر است؛ اعتبارسنجی آینده‌نگر لازم است.
\item \textbf{مدل‌ها}: \lr{ETHOS} از \lr{RF} ضعیف‌تر است؛ نتیجه‌ی منفی به‌صورت صادقانه گزارش شده است.
\item \textbf{تعمیم}: نتایج ممکن است به بیمارستان‌ها یا کشورهای دیگر منتقل نشود.
\item \textbf{سوگیری}: تفکیک سنی/جنسیتی انجام شده؛ ممیزی کامل انصاف توصیه می‌شود.
\end{itemize}
