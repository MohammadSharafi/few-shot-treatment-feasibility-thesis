\chapter{خلاصه‌ی معادلات کلیدی}
\label{app:equations}

این پیوست معادلات اصلی به‌کاررفته در پایان‌نامه را برای مرجع سریع گردآوری می‌کند.

%==============================================================================
\section{پیش‌پردازش داده}
%==============================================================================

نرمال‌سازی \lr{min-max} با برش به بازه‌ی بالینی:
\begin{equation}
x' = \frac{\text{\lr{clip}}(x, x_{\min}, x_{\max}) - x_{\min}}{x_{\max} - x_{\min}}
\end{equation}

%==============================================================================
\section{\texorpdfstring{توکن علائم}{توکن علائم}}
%==============================================================================

هر توکن: $\text{توکن} = [\text{\lr{symptom\_id}}, \text{\lr{intensity}}, \text{\lr{time\_delta}}, \text{\lr{modality}}]$

%==============================================================================
\section{شبکه‌های نمونه‌اولیه‌ای}
%==============================================================================

نمونه‌اولیه‌ی کلاس:
\begin{equation}
\mathbf{c}_k = \frac{1}{|\mathcal{S}_k|} \sum_{(x_i, y_i) \in \mathcal{S}_k} f_\theta(x_i)
\end{equation}

احتمال پرس‌وجو:
\begin{equation}
p(y = k \mid x) = \frac{\exp(-\langle f_\theta(x), \mathbf{c}_k \rangle / \tau)}{\sum_{k'} \exp(-\langle f_\theta(x), \mathbf{c}_{k'} \rangle / \tau)}
\end{equation}

%==============================================================================
\section{زیان کانونی}
%==============================================================================

\begin{equation}
\mathcal{L} = -\alpha (1-p_t)^\gamma \log(p_t)
\end{equation}
با $\gamma = 2$ و $\alpha$ از وزن‌های کلاس.

%==============================================================================
\section{\lr{FedAvg}}
%==============================================================================

\begin{equation}
\theta_{t+1} = \sum_{k=1}^{K} \frac{n_k}{n} \theta_{t+1}^{(k)}
\end{equation}

%==============================================================================
\section{معیارها}
%==============================================================================

\lr{AUROC}: $\int_0^1 \text{\lr{TPR}}(t) \, d(\text{\lr{FPR}}(t))$

\lr{F1} کلان: $\frac{F1_0 + F1_1}{2}$ که در آن $F1_k = \frac{2 \cdot P_k \cdot R_k}{P_k + R_k}$

\lr{MCC}: $\frac{TP \cdot TN - FP \cdot FN}{\sqrt{(TP+FP)(TP+FN)(TN+FP)(TN+FN)}}$

%==============================================================================
\section{ویژگی‌های مهندسی‌شده}
%==============================================================================

شاخص شوک: $\text{\lr{SI}} = \text{\lr{HR}} / \text{\lr{SBP}}$ (نرمال‌شده)

پراکسی \lr{SOFA}: $\sum_{j} w_j (1 - x_j)$ برای کراتینین، پلاکت، \lr{SpO2}، لاکتات

تعداد آزمایش غیرطبیعی: $\frac{1}{n} \sum_{i} \mathbb{1}[|x_i - 0.5| > 0.35]$

\section{خودتوجه}
%==============================================================================

$\text{\lr{Attention}}(Q,K,V) = \text{\lr{softmax}}(QK^\top / \sqrt{d_k}) V$

چندسر: $\text{\lr{MultiHead}} = \text{\lr{Concat}}(\text{\lr{head}}_1, \ldots, \text{\lr{head}}_h) W^O$
