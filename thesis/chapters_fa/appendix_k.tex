این پیوست عناصر خلاقانه و تجسم‌های تکمیلی را توصیف می‌کند.

%==============================================================================
\section{پوسترهای معماری}
%==============================================================================

\textbf{خط لوله‌ی \lr{ETHOS} (\pnum{3} نما):}
\begin{enumerate}
\item \textbf{جریان داده}: \lr{MIMIC} خام $\to$ کوهورت $\to$ آزمایش‌ها/علائم حیاتی $\to$ توکنها $\to$ دنباله‌ها.
\item \textbf{جریان مدل}: توکنها $\to$ رمزگذار \lr{ETHOS} $\to$ بردارهای نهفته $\to$ نمونه‌اولیه‌ها $\to$ پیش‌بینی.
\item \textbf{جریان فدرال}: آموزش محلی $\to$ بارگذاری گرادیان $\to$ \lr{FedAvg} $\to$ پخش مدل سراسری.
\end{enumerate}

%==============================================================================
\section{اینفوگرافیک سفر بیمار}
%==============================================================================

\textbf{روز \pnum{1} \lr{ICU} $\to$ \lr{Tokens} $\to$ پیش‌بینی $\to$ ترخیص:}
\begin{itemize}
\item ساعت \prange{0}{24}: آزمایش‌ها و علائم حیاتی جمع‌آوری می‌شوند.
\item \textbf{توکن‌سازی}: هر اندازه‌گیری $\to$ جفت (بین، شدت).
\item دنباله: \pnum{25} توکن به‌ازای هر بیمار (در صورت نیاز با \lr{padding}).
\item مدل: \lr{ETHOS} کدگذاری می‌کند $\to$ مقایسه با نمونه‌اولیه $\to$ ممکن/ناممکن.
\item پیامد: ترخیص به منزل/توان‌بخشی (ممکن) در برابر \lr{SNF}/مرگ (ناممکن).
\end{itemize}

\section{دموی تعاملی}
%==============================================================================

\textbf{برنامه‌ی \lr{Gradio} برای بالینیان:} یک رابط وب امکان بارگذاری \lr{CSV} ناشناس آزمایش/علامت حیاتی، توکن‌سازی لحظه‌ای، و نمایش پیش‌بینی‌های \lr{RF} + \lr{ETHOS} با توضیحات \lr{SHAP} را فراهم می‌کند. تنظیم آستانه و تحلیل حساسیت را پشتیبانی می‌کند.

%==============================================================================
\section{نمودار گردباد حساسیت}
%==============================================================================

نمودار گردباد حساسیت \lr{AUROC} به تغییر ابرپارامترها را نشان می‌دهد: \lr{max\_depth}، \lr{n\_estimators}، نرخ یادگیری، آستانه. \lr{RF} بیشترین حساسیت به آستانه دارد؛ \lr{ETHOS} به دما و $K$.
